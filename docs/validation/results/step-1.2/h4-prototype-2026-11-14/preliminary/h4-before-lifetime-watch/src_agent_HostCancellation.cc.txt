struct Watch {HWND window=nullptr;DWORD host=0,selected=0,count=0;bool bad=false;};
thread_local Watch *activeWatch=nullptr;
void CALLBACK event(HWINEVENTHOOK,DWORD,HWND window,LONG object,LONG child,DWORD tid,DWORD){
 if(!activeWatch||window!=activeWatch->window||object!=OBJID_WINDOW||child!=CHILDID_SELF)return;
 if(++activeWatch->count>256){activeWatch->bad=true;return;}

 if(thread&&GetProcessIdOfThread(thread.get())==activeWatch->host){if(activeWatch->selected&&activeWatch->selected!=tid)activeWatch->bad=true;activeWatch->selected=tid;}
}
DWORD discover(HWND window,DWORD pid){

  watch.selected=0;const auto name=L"VT7Pty host initialization "+std::to_wstring(GetTickCount64())+L" "+std::to_wstring(pass);
  host::require(SetConsoleTitleW(name.c_str())!=FALSE,L"Host title discovery");const auto end=GetTickCount64()+250;
  while(GetTickCount64()<end){MSG m;while(PeekMessageW(&m,nullptr,0,0,PM_REMOVE)){TranslateMessage(&m);DispatchMessageW(&m);}Sleep(1);}

}

void HostCancellation::cancel(){
 host::require(ready_&&!disabled_,L"Host cancellation unavailable");
 CONSOLE_SELECTION_INFO selection={};
 host::require(!console_.frozen()&&GetConsoleSelectionInfo(&selection)&&selection.dwFlags==0,L"Cancellation refused while console selected");
 host::require(ResetEvent(completion_.get())!=FALSE,L"Reset host keyboard boundary"); // All scraping and resize run on this same agent thread.
 try{
  const auto r=request(cancel::Operation::Cancel);
  host::require(WaitForSingleObject(completion_.get(),1000)==WAIT_OBJECT_0,L"OutcomeUnknown: keyboard completion boundary");
  const auto &k=r.keyboard;
  TRACE_EVENT(TraceSeverity::Info,TraceSubsystem::Console,"host-cancel result sequence=%llu result=%u host=%u thread=%u captured=%u applied=%u sent=%u restored=%u equal=%u error=%u",
   sequence_,static_cast<unsigned>(r.result),r.identity.host,r.identity.hostThread,k.captured,k.applied,k.sent,k.restored,k.equal,k.error);
  TRACE_EVENT(TraceSeverity::Info,TraceSubsystem::Console,"host-cancel before sequence=%llu keys=%s",sequence_,hex(k.before).c_str());
  TRACE_EVENT(TraceSeverity::Info,TraceSubsystem::Console,"host-cancel during sequence=%llu keys=%s",sequence_,hex(k.during).c_str());
  TRACE_EVENT(TraceSeverity::Info,TraceSubsystem::Console,"host-cancel after sequence=%llu keys=%s",sequence_,hex(k.after).c_str());
  host::require(r.result==cancel::Result::Accepted&&k.captured&&k.applied&&k.sent&&k.restored&&k.equal&&!k.error,L"Host cancellation or keyboard restoration failed");
 }catch(...){disabled_=true;ready_=false;TRACE_EVENT(TraceSeverity::Error,TraceSubsystem::Console,"host-cancel unavailable sequence=%llu outcome=unknown retry=disabled",sequence_);throw;}
}
void HostCancellation::cleanup() noexcept{
 if(hook_){UnhookWindowsHookEx(hook_);hook_=nullptr;}if(module_){FreeLibrary(module_);module_=nullptr;}if(atom_){GlobalDeleteAtom(atom_);atom_=0;}
 pipe_.reset();completion_.reset();thread_.reset();host_.reset();ready_=false;
}
void HostCancellation::close() noexcept{if(ready_&&!disabled_)try{const auto r=request(cancel::Operation::Close);(void)r;}catch(...){disabled_=true;}ready_=false;cleanup();}
HostCancellation::~HostCancellation(){close();}
