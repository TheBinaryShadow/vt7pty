[CmdletBinding()]
param([string]$Repository='D:\Git\vt7pty',[string]$Output='')
$ErrorActionPreference='Stop'
Set-StrictMode -Version 2.0
function Hash-Text([string]$Text) {
    $sha=[Security.Cryptography.SHA256]::Create()
    try { return [BitConverter]::ToString($sha.ComputeHash([Text.Encoding]::UTF8.GetBytes($Text))).Replace('-','').ToLowerInvariant() }
    finally { $sha.Dispose() }
}
$base=Join-Path $Repository 'docs\validation\results\step-1.2\host-cancellation-rc1'
$raw=Join-Path $base 'windows7'
$rows=@(); $caseRows=@()
foreach($machine in Get-ChildItem -LiteralPath $raw -Directory | Sort-Object Name) {
    $report=Get-Content -LiteralPath (Join-Path $machine.FullName 'report.json') -Raw | ConvertFrom-Json
    foreach($run in Get-ChildItem -LiteralPath $machine.FullName -Directory | Sort-Object Name) {
        $cases=Get-Content -LiteralPath (Join-Path $run.FullName 'suite\cases.json') -Raw | ConvertFrom-Json
        foreach($case in $cases.Cases) {
            $caseRows+=@([ordered]@{Machine=$report.ComputerName;Run=$run.Name;Case=$case.Case;Passed=$case.Passed})
            $dir=Join-Path $run.FullName ('suite\'+$case.Case)
            foreach($file in Get-ChildItem -LiteralPath $dir -Filter '*-request.json' | Sort-Object Name) {
                $q=Get-Content -LiteralPath $file.FullName -Raw | ConvertFrom-Json
                if($q.RequestOperation -ne 5 -or $q.Response.Result -ne 1) { continue }
                $k=$q.Response.Keyboard
                if($k.Before.Length -ne 512 -or $k.During.Length -ne 512 -or $k.After -cne $k.Before) { throw 'Invalid keyboard observation' }
                $rows+=@([ordered]@{
                    Machine=$report.ComputerName;RunId=$report.RunId;Run=$run.Name;Case=$case.Case;CasePassed=$case.Passed
                    RequestPath=$file.FullName.Substring($base.Length+1).Replace('\','/')
                    RequestSequence=$q.RequestSequence;RequestDeadline=$q.RequestDeadline
                    Before=$k.Before;During=$k.During;After=$k.After
                    DuringSha256=Hash-Text $k.During
                    KeyMessageSent=$k.Sent;KeyboardError=$k.Error
                })
            }
        }
    }
}
# Convert dictionaries to ordinary objects before Group-Object/Select-Object.
$records=@($rows | ForEach-Object {[pscustomobject]$_})
$caseRecords=@($caseRows | ForEach-Object {[pscustomobject]$_})
$groups=@($records | Group-Object Machine,During | ForEach-Object {
    [ordered]@{Machine=$_.Group[0].Machine;Requests=$_.Count;DuringSha256=$_.Group[0].DuringSha256
        FailedCaseRequests=@($_.Group | Where-Object {-not $_.CasePassed}).Count
        BeforeDistinct=@($_.Group.Before | Sort-Object -Unique).Count
        DuringDistinct=@($_.Group.During | Sort-Object -Unique).Count}
})
$matched=@(foreach($failure in $records | Where-Object {-not $_.CasePassed}) {
    $matches=@($records | Where-Object {$_.Machine -eq $failure.Machine -and $_.CasePassed -and
        $_.Before -ceq $failure.Before -and $_.During -ceq $failure.During -and $_.After -ceq $failure.After})
    [ordered]@{Machine=$failure.Machine;Run=$failure.Run;Case=$failure.Case;FailureRequest=$failure.RequestPath
        IdenticalBeforeDuringAfterPassingRequests=$matches.Count
        SameCasePassingRequests=@($matches | Where-Object {$_.Case -eq $failure.Case}).Count
        MatchingPaths=@($matches.RequestPath)}
})
$streams=@()
foreach($spec in @(@('esu-20261113-015305-c9e13b33','powershell-empty'),@('nonesu-20261113-015345-57e8a0fc','completed-before-cancel'))) {
    $items=@()
    foreach($n in 1..3) {
        $path=Join-Path $raw ($spec[0]+'\run-'+$n+'\suite\'+$spec[1]+'\stream.bin')
        $bytes=[IO.File]::ReadAllBytes($path)
        $text=[Text.UTF8Encoding]::new($false,$true).GetString($bytes)
        $normalized=[regex]::Replace($text,'\x1b\]0;VT7Pty dispatch research [0-9]+\x07','')
        $normalized=$normalized.Replace(('\run-'+$n+'\'),'\run-N\')
        $items+=@([pscustomobject]@{Run=$n;Path=$path.Substring($base.Length+1).Replace('\','/');Bytes=$bytes.Length;NormalizedSha256=Hash-Text $normalized})
    }
    $streams+=@([ordered]@{RunId=$spec[0];Case=$spec[1];DistinctNormalizedStreams=@($items.NormalizedSha256 | Sort-Object -Unique).Count;Streams=$items})
}
$caseTotals=@($caseRecords | Group-Object Case | ForEach-Object {
    [ordered]@{Case=$_.Name;Cases=$_.Count;Pass=@($_.Group | Where-Object {$_.Passed}).Count;Fail=@($_.Group | Where-Object {-not $_.Passed}).Count}
})
$result=[ordered]@{
    SchemaVersion=1;RecordedDate='2026-11-13';EvidenceSource='H3 rc1 preserved Windows 7 results'
    AcceptedCancelRequests=$records.Count;AssociatedWithFailedCases=@($records | Where-Object {-not $_.CasePassed}).Count
    Groups=$groups;FailureMatches=$matched;CaseTotals=$caseTotals
    StreamNormalization='Remove only VT OSC title records matching VT7Pty dispatch research <digits>; replace the literal per-repetition \run-N\ path component. Preserve all other VT bytes.'
    StreamComparisons=$streams;Requests=$rows
    Limits='Associates request with the containing case outcome. It does not invent event timing, internal host state, or a confirmed cause. The raw-input guard exercises the original runtime input transport, not the H3 key-message operation.'
}
if($records.Count -ne 88 -or $matched.Count -ne 5 -or $caseRecords.Count -ne 108) { throw 'Unexpected source inventory' }
if($Output) { [IO.File]::WriteAllText([IO.Path]::GetFullPath($Output),($result | ConvertTo-Json -Depth 12),[Text.UTF8Encoding]::new($false)) }
[pscustomobject]@{AcceptedCancelRequests=$records.Count;FailureComparisons=$matched.Count;Cases=$caseRecords.Count;KeyboardGroups=$groups.Count;StreamComparisons=$streams.Count} | ConvertTo-Json
