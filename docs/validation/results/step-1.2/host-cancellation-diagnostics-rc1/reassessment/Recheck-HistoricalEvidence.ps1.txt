param([Parameter(Mandatory=$true)][string]$RepositoryRoot)
$ErrorActionPreference='Stop'
$RepositoryRoot=[IO.Path]::GetFullPath($RepositoryRoot).TrimEnd('\')
$record=Get-Content -LiteralPath (Join-Path $RepositoryRoot 'docs\validation\results\step-1.2\host-cancellation-diagnostics-rc1\reassessment\historical-inventory-recheck.json') -Raw | ConvertFrom-Json
$total=0
foreach($set in $record.Sets){
    $base=[IO.Path]::GetFullPath((Join-Path $RepositoryRoot $set.Base))
    $inventory=Get-Content -LiteralPath (Join-Path $base $set.Inventory) -Raw | ConvertFrom-Json
    $files=@($inventory.($set.Property))
    if($files.Count -ne $set.Checked){throw ('Inventory count changed: '+$set.Base)}
    foreach($entry in $files){
        $path=[IO.Path]::GetFullPath((Join-Path $base $entry.Path))
        if(-not $path.StartsWith($base+'\',[StringComparison]::OrdinalIgnoreCase) -or -not (Test-Path -LiteralPath $path -PathType Leaf)){throw ('Missing or out-of-scope path: '+$entry.Path)}
        if((Get-Item -LiteralPath $path).Length -ne $entry.Bytes){throw ('Length mismatch: '+$entry.Path)}
        if((Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant() -cne $entry.Sha256){throw ('SHA-256 mismatch: '+$entry.Path)}
        $total++
    }
    Write-Output ($set.Base+': '+$files.Count+' matched')
}
if($total -ne $record.InventoryReferencesChecked){throw 'Overall count changed'}
Write-Output ('Matched '+$total+' inventory references. No native tests were run.')
