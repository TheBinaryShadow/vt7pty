std::wstring wide(std::string_view value) {

    check(n>0,"profile UTF-8 source");std::wstring result(n,0);
    check(MultiByteToWideChar(CP_UTF8,MB_ERR_INVALID_CHARS,value.data(),static_cast<int>(value.size()),result.data(),n)==n,"profile source conversion");return result;
}
std::wstring replace(std::wstring value,const std::wstring &from,const std::wstring &to) {
    size_t at=0;while((at=value.find(from,at))!=std::wstring::npos){value.replace(at,from.size(),to);at+=to.size();}return value;
}
std::wstring encoded(const std::wstring &value) {
    constexpr wchar_t table[]=L"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const auto *p=reinterpret_cast<const unsigned char *>(value.data());const size_t bytes=value.size()*2;
    std::wstring result;for(size_t i=0;i<bytes;i+=3){
        const unsigned n=(static_cast<unsigned>(p[i])<<16)|(i+1<bytes?static_cast<unsigned>(p[i+1])<<8:0)|(i+2<bytes?p[i+2]:0);
        result+=table[(n>>18)&63];result+=table[(n>>12)&63];result+=i+1<bytes?table[(n>>6)&63]:L'=';result+=i+2<bytes?table[n&63]:L'=';
    }return result;
}
uint64_t number(const std::string &value,const std::string &name) {
    auto at=value.find('"'+name+'"');check(at!=std::string::npos,"profile numeric field");at=value.find(':',at);check(at!=std::string::npos,"profile numeric delimiter");
    do{++at;}while(at<value.size() && isspace(static_cast<unsigned char>(value[at])));
    size_t end=at;while(end<value.size() && value[end]>='0' && value[end]<='9')++end;
    check(end>at,"profile positive numeric value");return std::stoull(value.substr(at,end-at));
}
bool truth(const std::string &value,const std::string &name) {
    auto at=value.find('"'+name+'"');if(at==std::string::npos)return false;at=value.find(':',at);if(at==std::string::npos)return false;
    do{++at;}while(at<value.size() && isspace(static_cast<unsigned char>(value[at])));return value.compare(at,4,"true")==0;
}

std::wstring flattened(std::wstring source) {
    std::vector<std::wstring> lines;size_t at=0;
    while(at<source.size()){
        const auto end=source.find(L'\n',at);auto line=source.substr(at,end==std::wstring::npos?source.size()-at:end-at);
        while(!line.empty() && (line.back()==L'\r'||line.back()==L' '))line.pop_back();
        const auto first=line.find_first_not_of(L" \t");
        if(first!=std::wstring::npos && line[first]!=L'#')lines.push_back(line.substr(first));
        if(end==std::wstring::npos)break;at=end+1;
    }
    std::wstring result;
    for(size_t i=0;i<lines.size();++i){
        result+=lines[i];
        const bool continued=i+1<lines.size() && (lines[i+1].starts_with(L"else")||lines[i+1].starts_with(L"catch")||lines[i+1].starts_with(L"finally"));
        if(lines[i].back()!=L'{' && lines[i].back()!=L';' && !continued)result+=L';';result+=L' ';
    }return result;
}

LONG volatile childControls=0;

void exercise(Context &c,const Spec &spec){
    auto &s=c.start();Shell shell(c,spec);shell.snapshot(0);
    const auto name=shell.checkName();std::string proofs="[";
    auto keep=[&](const std::wstring &which){if(proofs.size()>1)proofs+=',';proofs+=shell.proof(which);};
    s.input(utf8(name+L" Editing 'EDXT \u010d\u0107\u0111\u0161\u017e'"));
    for(int n=0;n<8;++n)s.input("\x1b[D");s.input("\x7fI\x1b[F\r");
    check(s.until([&]{return shell.atPrompt() && GetFileAttributesW((c.out+L"\\Editing.json").c_str())!=INVALID_FILE_ATTRIBUTES;}),"profile editing result");keep(L"Editing");
    shell.command(name+L" History 'HISTORY_OK'");const auto first=number(shell.proof(L"History"),"Invocation");
    s.input("\x1b[A\r");check(s.until([&]{try{return shell.atPrompt() && number(fileText(c.out+L"\\History.json"),"Invocation")>first;}catch(...){return false;}}),"profile in-memory history recall");keep(L"History");
    shell.command(L"VT7PtyProfileComple\t");keep(L"Completion");
    s.input(utf8(name+L" Multiline @'\rfirst\u010d\rsecond\u017e\r'@\r\r"));
    check(s.until([&]{return GetFileAttributesW((c.out+L"\\Multiline.json").c_str())!=INVALID_FILE_ATTRIBUTES && shell.atPrompt();},15000),"profile multiline result");keep(L"Multiline");
    const auto before=shell.generation;
    s.input(utf8(name+L" Cancelled 'SHOULD_NOT_RUN'"));check(s.until([&]{return !shell.atPrompt();}),"profile partial line visibly pending");s.input("\x03");
    check(s.until([&]{return shell.atPrompt() && shell.generation>before;}),"profile cancellation preserves prompt function");s.input("\r");
    shell.command(name+L" Continue 'CONTINUE_OK'");keep(L"Continue");
    check(GetFileAttributesW((c.out+L"\\Cancelled.json").c_str())==INVALID_FILE_ATTRIBUTES,"profile canceled text never executes");
    writeFile(c.out+L"\\cancel.json","{\"PartialLineDiscarded\":true,\"NextCommandPassed\":true}");
    const auto previous=shell.generation;s.input(utf8(L"VT7PtyProfileChild_"+wide(shell.nonce)+L"\r"));
    check(s.until([&]{return GetFileAttributesW((c.out+L"\\child-ready.json").c_str())!=INVALID_FILE_ATTRIBUTES;}),"profile native child ready");shell.retainChild();s.input("\x03");

    shell.command(name+L" Resize 'RESIZE_OK'");keep(L"Resize");shell.snapshot(1);

    shell.quit();writeFile(c.out+L"\\command-proofs.json","{\"Proofs\":"+proofs+"]}");
    writeFile(c.out+L"\\profile-result.json","{\"ShellId\":"+std::to_string(shell.pid)+",\"NaturalExit\":37,\"WorkflowsPassed\":true}");
}
