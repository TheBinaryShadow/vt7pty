$ErrorActionPreference='Stop'
Set-StrictMode -Version 2.0
$repo='D:\Git\vt7pty'
$execution=Get-Content -LiteralPath (Join-Path $repo 'artifacts\fidelity\h3-coordination-exact-package-execution.json') -Raw | ConvertFrom-Json
$reviewPath=Join-Path $repo 'artifacts\fidelity\h3-coordination-exact-package-review.json'
if(-not (Test-Path -LiteralPath $reviewPath)) {$null=& (Join-Path $repo 'tools\research\Test-HostCoordinationEvidence.ps1') -Directory $execution.Results -PackageRoot $execution.Extraction -OutputPath $reviewPath}
$review=Get-Content -LiteralPath $reviewPath -Raw | ConvertFrom-Json
$report=Get-Content -LiteralPath (Join-Path $execution.Results 'report.json') -Raw | ConvertFrom-Json
$metrics=[ordered]@{RunId=$report.RunId;Collected=$report.Completed;Trials=@();Cases=0;CasesPassed=0;VerifiedEvidenceFiles=$review.VerifiedEvidenceFiles;
    PrototypeBaselineCases=0;PrototypeBaselinePassed=0;PrototypePhases=@();PrototypeOwnedScopes=0;PrototypeBorrowedScopes=0;ExpiryRefusals=0;
    PrototypeCoverageGaps=@();PrototypeOperationalIssues=@();ReferenceCoverageGaps=@();NeighborAgentTotals=@();NeighborCollectors=@()}
foreach($trial in $report.Trials) {
    $passed=@($trial.Cases.Cases | Where-Object {$_.Passed}).Count
    $metrics.Cases+=$trial.Cases.Cases.Count;$metrics.CasesPassed+=$passed
    $metrics.Trials+=@([ordered]@{Order=$trial.Order;Role=$trial.Role;Rotation=$trial.Rotation;Outcome=$trial.Outcome;CollectionStatus=$trial.CollectionStatus;Cases=$trial.Cases.Cases.Count;Passed=$passed;
        DurationMilliseconds=$trial.DurationMilliseconds;OwnedLifetimesCleaned=$trial.Observation.OwnedLifetimesCleaned;DesktopIsolationVerified=$trial.Observation.DesktopIsolationVerified;
        FailedCases=@($trial.Cases.Cases | Where-Object {-not $_.Passed} | ForEach-Object {$_.Case})})
    $item=@($review.Trials | Where-Object {$_.Order -eq $trial.Order})[0]
    if($trial.Role -eq 'Reference') {$metrics.ReferenceCoverageGaps+=@($item.Review.CoverageGaps)+@($item.Review.Core.CoverageGaps)}
    if($trial.Role -ne 'Experiment') {continue}
    $baseline=@($trial.Cases.Cases | Where-Object {$_.Case -notlike 'focus-*'})
    $metrics.PrototypeBaselineCases+=$baseline.Count;$metrics.PrototypeBaselinePassed+=@($baseline | Where-Object {$_.Passed}).Count
    $metrics.PrototypePhases+=@($item.Review.Conditions);$metrics.PrototypeOwnedScopes+=$item.Review.Coordination.OwnedScopes;$metrics.PrototypeBorrowedScopes+=$item.Review.Coordination.BorrowedScopes
    $metrics.ExpiryRefusals+=$item.Review.Coordination.ExpiryRefusals;$metrics.NeighborCollectors+=@($item.Review.Coordination.NeighborCollector)
    $metrics.PrototypeCoverageGaps+=@($item.Review.CoverageGaps)+@($item.Review.Core.CoverageGaps);$metrics.PrototypeOperationalIssues+=@($item.Review.OperationalIssues)
    $finish=Get-Content -LiteralPath (Join-Path $execution.Results ($trial.Directory+'\native\neighbor\agent-finish.json')) -Raw | ConvertFrom-Json
    $metrics.NeighborAgentTotals+=@($finish.LastTotal)
}
if($metrics.Collected -ne 9 -or $metrics.Cases -ne 171 -or $metrics.PrototypeBaselineCases -ne 42 -or $metrics.PrototypeBaselinePassed -ne 42 -or
    $metrics.ExpiryRefusals -ne 3 -or $metrics.PrototypeCoverageGaps.Count -or $metrics.PrototypeOperationalIssues.Count) {throw 'Inspect failed or incomplete prototype verification.'}
[IO.File]::WriteAllText((Join-Path $repo 'artifacts\fidelity\h3-coordination-exact-package-metrics.json'),($metrics | ConvertTo-Json -Depth 14),[Text.UTF8Encoding]::new($false))
[pscustomobject]@{RunId=$metrics.RunId;Collected=$metrics.Collected;Cases=$metrics.Cases;CasesPassed=$metrics.CasesPassed;BaselinePassed=$metrics.PrototypeBaselinePassed;Hashes=$metrics.VerifiedEvidenceFiles;
    PrototypeGaps=$metrics.PrototypeCoverageGaps.Count;PrototypeIssues=$metrics.PrototypeOperationalIssues.Count}
