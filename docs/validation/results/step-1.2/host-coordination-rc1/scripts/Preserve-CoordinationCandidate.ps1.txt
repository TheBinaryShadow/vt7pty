$package=Join-Path $repo 'artifacts\fidelity\packages\VT7Pty-host-coordination-0.5.0-rc1'
$manifest=Get-Content -LiteralPath (Join-Path $package 'manifest.json') -Raw | ConvertFrom-Json
$records=New-Object 'Collections.Generic.List[object]'
function Copy-Exact([string]$Source,[string]$Relative) {
    $destination=[IO.Path]::GetFullPath((Join-Path $target $Relative))
    if(-not $destination.StartsWith($target+'\',[StringComparison]::OrdinalIgnoreCase)) {throw 'Evidence target leaves its directory.'}
    [IO.Directory]::CreateDirectory((Split-Path -Parent $destination)) | Out-Null
    $hash=(Get-FileHash -LiteralPath $Source -Algorithm SHA256).Hash.ToLowerInvariant()
    if(Test-Path -LiteralPath $destination) {if((Get-FileHash -LiteralPath $destination -Algorithm SHA256).Hash.ToLowerInvariant() -cne $hash) {throw 'Different evidence already exists.'}}
    else {[IO.File]::Copy($Source,$destination)}
    if((Get-Item -LiteralPath $destination).Length -ne (Get-Item -LiteralPath $Source).Length -or (Get-FileHash -LiteralPath $destination -Algorithm SHA256).Hash.ToLowerInvariant() -cne $hash) {throw 'Evidence copy differs.'}
    $records.Add([ordered]@{Path=$Relative.Replace('\','/');Bytes=(Get-Item -LiteralPath $destination).Length;Sha256=$hash})
}
function Copy-Tree([string]$Source,[string]$Relative,[bool]$SkipPrograms=$false) {
    $sourcePath=[IO.Path]::GetFullPath($Source).TrimEnd('\')
    foreach($file in @(Get-ChildItem -LiteralPath $sourcePath -File -Recurse)) {
        if($SkipPrograms -and $file.Extension -in @('.exe','.dll','.pdb','.lib','.obj','.exp')) {continue}
        Copy-Exact $file.FullName (Join-Path $Relative $file.FullName.Substring($sourcePath.Length+1))
    }
}
