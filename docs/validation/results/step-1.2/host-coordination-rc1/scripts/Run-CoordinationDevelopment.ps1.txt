param([string]$Configuration='Debug',[string]$Name='first-debug',[int]$Rotation=0,[int]$SoakMs=0)
$ErrorActionPreference='Stop'
Set-StrictMode -Version 2.0
$repo='D:\Git\vt7pty'
$output=Join-Path $repo ('artifacts\fidelity\h3-coordination-'+$Name)
if(Test-Path -LiteralPath $output) {throw 'Use a fresh development run name.'}
$bin=Join-Path $output 'bin';$native=Join-Path $output 'native'
[IO.Directory]::CreateDirectory($bin) | Out-Null
[IO.Directory]::CreateDirectory($native) | Out-Null
foreach($name in @('HostCancellationCoordinatedProbe.exe','HostCancellationCoordinatedProbe.pdb','HostCancellationCoordinatedHook.dll','HostCancellationCoordinatedHook.pdb')) {
    Copy-Item -LiteralPath (Join-Path $repo ('artifacts\bin\x64\'+$Configuration+'\'+$name)) -Destination $bin
}
Copy-Item -LiteralPath (Join-Path $repo ('artifacts\bin\x64\'+$Configuration+'\VT7Pty-CoordinatedAgent.exe')) -Destination (Join-Path $bin 'VT7Pty-Agent.exe')
Copy-Item -LiteralPath (Join-Path $repo 'artifacts\fidelity\packages\VT7Pty-host-freeze-diagnostics-0.5.0-rc1\control\bin\VT7Pty.dll') -Destination $bin
$start=New-Object Diagnostics.ProcessStartInfo
$start.FileName=Join-Path $bin 'HostCancellationCoordinatedProbe.exe';$start.WorkingDirectory=$bin;$start.Arguments='"'+$native+'"'
$start.UseShellExecute=$false;$start.CreateNoWindow=$true;$start.RedirectStandardOutput=$true;$start.RedirectStandardError=$true
$start.EnvironmentVariables['VT7PTY_H3_ORDER']=[string]$Rotation
$start.EnvironmentVariables['VT7PTY_H3_NEIGHBOR_SOAK_MS']=[string]$SoakMs
$p=New-Object Diagnostics.Process;$p.StartInfo=$start
if(-not $p.Start()) {throw 'Cannot start the created research controller.'}
$stdout=$p.StandardOutput.ReadToEndAsync();$stderr=$p.StandardError.ReadToEndAsync()
if(-not $p.WaitForExit(500000)) {$p.Kill();$p.WaitForExit();throw 'Created controller exceeded its deadline.'}
[IO.File]::WriteAllText((Join-Path $output 'stdout.txt'),$stdout.Result,[Text.UTF8Encoding]::new($false))
[IO.File]::WriteAllText((Join-Path $output 'stderr.txt'),$stderr.Result,[Text.UTF8Encoding]::new($false))
$root=Get-Content -LiteralPath (Join-Path $native 'probe.json') -Raw | ConvertFrom-Json
$cases=Get-Content -LiteralPath (Join-Path $native 'suite\cases.json') -Raw | ConvertFrom-Json
$result=[ordered]@{NativeExit=$p.ExitCode;NativeStatus=$root.Status;Cases=$cases.Cases.Count;FailedCases=@($cases.Cases | Where-Object {-not $_.Passed});Root=$root;
    SourceCommit=(& git -C $repo rev-parse HEAD).Trim();SourceTreeClean=(@(& git -C $repo status --porcelain=v1).Count -eq 0);Configuration=$Configuration;Rotation=$Rotation;NeighborSoakMs=$SoakMs}
[IO.File]::WriteAllText((Join-Path $output 'execution.json'),($result | ConvertTo-Json -Depth 8),[Text.UTF8Encoding]::new($false))
$p.Close();$result | ConvertTo-Json -Depth 8
