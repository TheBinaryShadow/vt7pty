std::string hex(std::string_view bytes) {
    std::string s;constexpr char digit[]="0123456789abcdef";
    for(unsigned char c:bytes){s+=digit[c>>4];s+=digit[c&15];}return s;
}
std::string specJson(const Spec &s) {
    return "{\"Case\":"+json(s.name)+",\"Kind\":"+json(s.kind)+",\"Mode\":"+std::to_string(s.mode)+
        ",\"InputHex\":"+json(hex(s.input))+",\"ExpectedText\":"+json(utf8(s.expected))+
        ",\"SecondText\":"+json(utf8(s.second))+",\"ChunkBytes\":"+std::to_string(s.chunk)+
        ",\"GapMilliseconds\":"+std::to_string(s.gap)+",\"LiveReplies\":"+boolean(s.liveReplies)+"}";
}
std::string listJson() {
    std::string s="{\"SchemaVersion\":1,\"Cases\":[";bool first=true;
    for(const auto &v:specs()){if(!first)s+=',';first=false;s+=specJson(v);}return s+"]}";
}
void validate(const Packet &p) {
    check(p.magic==0x31464e49 && p.schema==1 && p.pid && p.mode<=3 && p.recordCount<=1024 &&
        p.textCount<=2047 && !p.overflow,"bounded input capture packet");
}
std::string packet(const Packet &p) {
    validate(p);std::string records="[";
    for(DWORD i=0;i<p.recordCount;++i){if(i)records+=',';const auto &r=p.records[i];
        records+="{\"Type\":"+std::to_string(r.type)+",\"Down\":"+std::to_string(r.down)+
            ",\"Repeat\":"+std::to_string(r.repeat)+",\"VirtualKey\":"+std::to_string(r.vk)+
            ",\"ScanCode\":"+std::to_string(r.scan)+",\"Character\":"+std::to_string(r.character)+
            ",\"Modifiers\":"+std::to_string(r.modifiers)+",\"X\":"+std::to_string(r.x)+",\"Y\":"+std::to_string(r.y)+
            ",\"Buttons\":"+std::to_string(r.buttons)+",\"Flags\":"+std::to_string(r.flags)+"}";
    }
    return "{\"SchemaVersion\":1,\"ProcessId\":"+std::to_string(p.pid)+",\"Mode\":"+std::to_string(p.mode)+
        ",\"InputMode\":"+std::to_string(p.inputMode)+",\"Overflow\":"+std::to_string(p.overflow)+
        ",\"ReadStarted\":"+std::to_string(p.readStarted)+",\"ReadReturned\":"+std::to_string(p.readReturned)+

        ",\"Text\":"+json(utf8(std::wstring(p.text,p.textCount)))+

        ",\"ChildExited\":"+std::to_string(p.referenceChildExited)+"},\"Records\":"+records+"]}";

struct Capture {
    Session &s;std::wstring out;bool active=false;
    std::unique_ptr<Packet> p=std::make_unique<Packet>();
    Capture(Session &session,const std::wstring &path,uint32_t mode):s(session),out(path) {
        char command='V';s.inputCaptureTransfer(&command,1,true);s.inputCaptureTransfer(&mode,sizeof(mode),true);
        active=true;receive();writeFile(out+L"\\capture-ready.json",packet(*p));
    }
    void receive(){s.inputCaptureTransfer(p.get(),sizeof(*p),false);validate(*p);}
    const Packet &get(char command='S'){s.inputCaptureTransfer(&command,1,true);receive();return *p;}
    bool until(const std::function<bool(const Packet &)> &test,DWORD ms=5000){
        return s.until([&]{return test(get());},ms);
    }
    void close() {
        if(!active)return;get('Q');active=false;
        writeFile(out+L"\\capture-stopped.json",packet(*p));
        check(!p->stopCanceledRead,"capture finished without canceled fixture read");
    }
    ~Capture(){if(active){try{close();}catch(...){}}}
};
std::vector<inputfidelity::Record> keys(const Packet &p,bool includeMarker=false) {
    std::vector<inputfidelity::Record> v;
    for(DWORD i=0;i<p.recordCount;++i){const auto &r=p.records[i];
        if(r.type!=KEY_EVENT || !r.down || r.vk==VK_SHIFT || r.vk==VK_CONTROL || r.vk==VK_MENU)continue;
        if(!includeMarker && r.character==0xe000)continue;
        check(r.repeat>0 && r.repeat<=1024,"positive bounded repeat");
        for(DWORD n=0;n<r.repeat;++n)v.push_back(r);
    }
    return v;
}
std::wstring text(const Packet &p) {
    std::wstring s;for(const auto &r:keys(p))s+=static_cast<wchar_t>(r.character);return s;
}
bool marker(const Packet &p) {
    for(DWORD i=0;i<p.recordCount;++i)if(p.records[i].type==KEY_EVENT && !p.records[i].down && p.records[i].character==0xe000)return true;
    return false;
}
void keyStateReleased(const Packet &p) {
    std::array<int,256> state={};
    for(DWORD i=0;i<p.recordCount;++i){const auto &r=p.records[i];if(r.type!=KEY_EVENT)continue;
        check(r.vk<state.size() && r.repeat>0,"key record bound");
        state[r.vk]+=r.down?static_cast<int>(r.repeat):-static_cast<int>(r.repeat);
        check(state[r.vk]>=0,"key-up follows key-down");
    }
    check(std::all_of(state.begin(),state.end(),[](int value){return value==0;}),"all generated key states released");
}
void rawOracle(const Spec &s,const Packet &p) {
    const auto v=keys(p);
    check(p.c==(s.kind=="processed"?1u:0u) && p.brk==0,"distinct control count");
    if(s.kind=="navigation"){
        const std::vector<DWORD> expected={VK_LEFT,VK_RIGHT,VK_UP,VK_DOWN,VK_HOME,VK_END,VK_DELETE,VK_INSERT,VK_PRIOR,VK_NEXT,'Z'};
        check(v.size()==expected.size(),"exact navigation count");
        for(size_t i=0;i<v.size();++i){check(v[i].vk==expected[i],"navigation order");
            if(i+1<v.size())check(v[i].character==0 && v[i].modifiers==ENHANCED_KEY,"enhanced navigation record");
            else check(v[i].character=='z' && !v[i].modifiers,"ordinary next key after navigation");
        }
    }else if(s.kind=="modifiers"){
        const std::vector<DWORD> vk={VK_LEFT,VK_RIGHT,VK_HOME,'A','Z'},mods={8,16,2,2,0};
        check(v.size()==vk.size(),"modifier count");
        const std::vector<DWORD> characters={0,0,0,97,122};
        for(size_t i=0;i<v.size();++i)check(v[i].vk==vk[i] && v[i].character==characters[i] &&
            v[i].modifiers==(mods[i]+(i<3?ENHANCED_KEY:0u)),"exact chord and released next key");
    }else if(s.kind=="alt"){
        check(v.size()==2 && v[0].character=='a' && (v[0].modifiers&LEFT_ALT_PRESSED) && v[1].character=='z' && !(v[1].modifiers&31),"fragmented Alt text");
    }else if(s.kind=="mouse" || s.kind=="x10"){
        std::vector<inputfidelity::Record> m;
        for(DWORD i=0;i<p.recordCount;++i)if(p.records[i].type==MOUSE_EVENT)m.push_back(p.records[i]);
        const std::vector<DWORD> buttons=s.kind=="mouse"?std::vector<DWORD>{1,1,0,0x780000,0xff880000,1,0}:std::vector<DWORD>{1,0};
        const std::vector<DWORD> flags=s.kind=="mouse"?std::vector<DWORD>{0,1,0,4,4,0,0}:std::vector<DWORD>{0,0};
        const std::vector<int> x=s.kind=="mouse"?std::vector<int>{9,10,10,11,11,19,19}:std::vector<int>{9,9};
        const std::vector<int> y=s.kind=="mouse"?std::vector<int>{4,5,5,6,6,7,7}:std::vector<int>{4,4};
        check(m.size()==buttons.size() && v.empty(),"exact mouse-only delivery");
        for(size_t i=0;i<m.size();++i)check(m[i].buttons==buttons[i] && m[i].flags==flags[i] &&
            m[i].x==static_cast<int>(p.windowLeft)+x[i] && m[i].y==static_cast<int>(p.windowTop)+y[i] &&
            m[i].modifiers==(s.kind=="mouse" && i==5 ? 24u:0u),"mouse coordinate, button, wheel and modifier");
    }else if(s.kind=="reference"){
        check(p.reference.pid && p.reference.pid==p.referenceGroup && p.referenceChildExited &&
            p.referenceExit==37 && p.reference.brk==1 && p.reference.c==0 && v.empty(),"native Break reference only");
        check(static_cast<bool>(p.reference.inputMode&ENABLE_PROCESSED_INPUT)==(s.mode==2),"native Break mode");
    }else{
        check(text(p)==s.expected,"exact text order and multiplicity");
        for(DWORD i=0;i<p.recordCount;++i)check(p.records[i].type!=MOUSE_EVENT,"no stray mouse event");
    }
    keyStateReleased(p);
}
void send(Session &s,const Spec &spec,const std::wstring &out) {
    std::string chunks="[";size_t count=0;
    const size_t chunk=spec.chunk?spec.chunk:std::max<size_t>(1,spec.input.size());
    for(size_t i=0;i<spec.input.size();i+=chunk){
        const auto bytes=spec.input.substr(i,chunk);s.input(bytes);
        if(count++)chunks+=',';chunks+=json(hex(bytes));
        if(i+chunk<spec.input.size() && spec.gap){
            const auto until=GetTickCount64()+spec.gap;
            do{if(spec.liveReplies)s.pump();Sleep(2);}while(GetTickCount64()<until);
        }
    }
    writeFile(out+L"\\writes.json","{\"ChunkHex\":"+chunks+"],\"GapMilliseconds\":"+std::to_string(spec.gap)+"}");
}
void exercise(Context &c,const Spec &spec) {
    auto &s=c.start();s.inputReplyRecording=true;
    {
        const auto modeOffset=s.raw.size();
        Capture capture(s,c.out,spec.mode);
        check(capture.until([&](const Packet &p){return spec.mode!=1 || p.readStarted;}),"capture readiness");
        if(spec.mode==inputfidelity::Mouse){
            check(s.until([&]{return s.raw.find("\x1b[?1006h",modeOffset)!=std::string::npos;}),"agent mouse-mode negotiation");
            writeFile(c.out+L"\\mouse-ready.json","{\"Mode1006Observed\":true,\"StreamStartOffset\":"+std::to_string(modeOffset)+"}");
        }
        send(s,spec,c.out);
        DWORD observedRecords=0;
        if(spec.mode==1){
            if(spec.kind=="cancel"){
                s.input("\x03");
                check(capture.until([](const Packet &p){return p.readReturned && p.c==1;}),"Unicode cooked cancellation");
            }else check(capture.until([](const Packet &p){return p.readReturned!=0;}),"cooked read completion");
            writeFile(c.out+L"\\input-observed.json",packet(*capture.p));
            if(spec.kind=="cancel")check(capture.p->readError==995 && capture.p->textCount==0 && capture.p->c==1 && !capture.p->brk,"cooked Unicode line canceled exactly once");
            else check(capture.p->readSucceeded && !capture.p->readError && !capture.p->c && !capture.p->brk &&
                std::wstring(capture.p->text,capture.p->textCount)==spec.expected,"exact cooked editing/paste readback");
            if(spec.kind=="cancel" || spec.kind=="multiline"){
                capture.get('R');
                if(spec.kind=="cancel")s.input(utf8(L"NEXT \u017e\r"));
                check(capture.until([](const Packet &p){return p.readReturned!=0;}),"second cooked read");
                writeFile(c.out+L"\\second-read.json",packet(*capture.p));
                const auto expected=spec.kind=="cancel"?L"NEXT \u017e\r\n":spec.second;
                check(!capture.p->readError && capture.p->readSucceeded && std::wstring(capture.p->text,capture.p->textCount)==expected,"exact next queued line");
            }
        }else{
            if(spec.kind=="reference")capture.get('B');
            s.input("\xee\x80\x80");
            check(capture.until([&](const Packet &p){return marker(p) && (spec.kind!="processed" || p.c>=1);}),"input stream completion fence");
            // A second observed fence interval catches trailing records. It is
            // not an interactive latency benchmark.
            const auto quiet=GetTickCount64()+200;
            while(GetTickCount64()<quiet){s.pump();capture.get();Sleep(5);}
            writeFile(c.out+L"\\input-observed.json",packet(*capture.p));
            std::string replies="[";
            for(size_t i=0;i<s.inputTerminalReplies.size();++i){if(i)replies+=',';replies+=json(hex(s.inputTerminalReplies[i]));}
            writeFile(c.out+L"\\terminal-replies.json","{\"ReplyHex\":"+replies+"]}");
            rawOracle(spec,*capture.p);observedRecords=capture.p->recordCount;
        }
        capture.close();
        if(spec.mode!=1)check(capture.p->recordCount==observedRecords,"no late input records after completion fence");
    }
    nextRead(s,c.out,0);
}
void oracle() {
    auto storage=std::make_unique<Packet>();auto &p=*storage;p.pid=1;p.recordCount=2;
    p.records[0]={KEY_EVENT,1,1,'A',0,'a',0};
    p.records[1]={KEY_EVENT,0,1,'A',0,'a',0};
    check(text(p)==L"a","input text oracle");
    keyStateReleased(p);
    p.records[0].repeat=2;
    check(text(p)==L"aa","repeat normalization oracle");
    bool rejected=false;try{keyStateReleased(p);}catch(...){rejected=true;}
    check(rejected,"unbalanced input rejected");
    p.overflow=1;rejected=false;try{validate(p);}catch(...){rejected=true;}
    check(rejected,"overflow rejected");
    check(specs().size()==33,"complete input inventory");
    printf("Input fidelity oracle: 5 checks passed.\n");
}
int run(const std::wstring &out,const std::string &only="") {
    outRoot=out;check(CreateDirectoryW(out.c_str(),nullptr),"new input collection");rootEvidenceCreated=true;
    applicationDesktop=nameOf(GetThreadDesktop(GetCurrentThreadId()));
    if(!desktopPreflight(out+L"\\preflight.json"))return 2;
    check(SetEnvironmentVariableW(L"VT7PTY_DEBUG",L"trace"),"input runtime trace");
    Collector collector(out);std::string rows="[";size_t total=0,passed=0;
    for(const auto &spec:specs()){
        if(!only.empty() && spec.name!=only)continue;
        Context c(out+L"\\"+std::wstring(spec.name.begin(),spec.name.end()));
        writeFile(c.out+L"\\recipe.json",specJson(spec));
        bool okay=false;std::string error;
        try{check(desktopPreflight(c.out+L"\\preflight.json"),"input case eligible desktop");exercise(c,spec);okay=true;}
        catch(const std::exception &e){error=e.what();}
        okay=c.finish() && okay;
        if(error.empty() && !c.error.empty())error=c.error;
        if(total++)rows+=',';if(okay)++passed;
        rows+="{\"Case\":"+json(spec.name)+",\"Passed\":"+boolean(okay)+",\"OwnedProcessesCleaned\":"+boolean(c.cleaned)+
            ",\"HostExited\":"+boolean(c.hostExited)+",\"Error\":"+json(error)+"}";
        printf("%s: %s %s\n",okay?"PASS":"FAIL",spec.name.c_str(),error.c_str());
    }
    collector.stop();check(total>0,"input case selection");
    writeFile(out+L"\\cases.json","{\"SchemaVersion\":1,\"Cases\":"+rows+"]}");
    writeFile(out+L"\\probe.json","{\"Kind\":\"InputFidelityProbe\",\"SchemaVersion\":1,\"Status\":"+json(passed==total?"Pass":"Fail")+
        ",\"SourceCommit\":"+json(GenVersion_Commit)+",\"RuntimeSourceCommit\":"+json(manualRuntimeSource())+
        ",\"Passed\":"+std::to_string(passed)+",\"Total\":"+std::to_string(total)+
        ",\"FullInventory\":"+boolean(only.empty())+",\"CtrlBreakTransport\":\"Unsupported\",\"NativeBreakReferenceOnly\":true,\"Step12Accepted\":false}");
    return passed==total?0:1;
}
