def sha(b):return hashlib.sha256(b).hexdigest()
def dump(p,obj):p.write_text(json.dumps(obj,indent=2)+'\n',encoding='utf-8')
def archive(name,trees):
 records=[]
 with zipfile.ZipFile(out/name,'w',zipfile.ZIP_DEFLATED,compresslevel=6) as z:
  for tree in trees:
   for path in sorted(tree.rglob('*')):
    if not path.is_file():continue
    b=path.read_bytes();member=tree.name+'/'+path.relative_to(tree).as_posix()
    z.writestr(member,b);records.append(dict(Path=member,Bytes=len(b),Sha256=sha(b)))
 with zipfile.ZipFile(out/name) as z:
  assert len(z.infolist())==len(records)
  for item in records:
   b=z.read(item['Path']);assert len(b)==item['Bytes'] and sha(b)==item['Sha256']
 dump(out/(name+'.files.json'),dict(SchemaVersion=1,Files=records))
 return dict(Archive=name,Files=len(records),Bytes=(out/name).stat().st_size,Sha256=sha((out/name).read_bytes()))
