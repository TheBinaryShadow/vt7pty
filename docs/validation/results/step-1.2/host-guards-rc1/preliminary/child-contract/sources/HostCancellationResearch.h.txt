struct Query {
    LONG status=static_cast<LONG>(0xc0000002u);
    ULONG length=0;
    ULONG_PTR raw=0;
};
// phnt identifies ProcessConsoleHostProcess as class 49, ULONG_PTR.
// Internal interface: retain raw results and reject unfamiliar representations.
inline Query hostQuery(HANDLE process) {
    using Function=LONG (NTAPI *)(HANDLE,ULONG,void *,ULONG,ULONG *);
    const HMODULE module=GetModuleHandleW(L"ntdll.dll"); if(!module) return {};
    const auto address=GetProcAddress(module,"NtQueryInformationProcess");
    Function function=nullptr;
    static_assert(sizeof(function)==sizeof(address));
    memcpy(&function,&address,sizeof(function));
    Query result;
    if(function) result.status=function(process,49,&result.raw,sizeof(result.raw),&result.length);
    return result;
}
inline bool valid(const Query &q) {
    return q.status>=0 && q.length==sizeof(ULONG_PTR) && q.raw!=0 &&
        q.raw<=std::numeric_limits<DWORD>::max() && (q.raw&~ULONG_PTR{3})!=0;
}
inline bool agrees(const Query &a,const Query &b) { return valid(a) && valid(b) && a.raw==b.raw; }
inline std::string queryJson(const Query &q) {
    return "{\"NtStatus\":"+std::to_string(static_cast<ULONG>(q.status))+
        ",\"ReturnLength\":"+std::to_string(q.length)+",\"RawHostValue\":"+std::to_string(q.raw)+",\"LowBits\":"+std::to_string(q.raw&3u)+
        ",\"NormalizedPid\":"+std::to_string(q.raw&~ULONG_PTR{3})+"}";
}
inline std::string utf8(std::wstring_view text) {
    if(text.empty()) return {};
    check(text.size()<32768,"UTF-8 size");
    const int size=WideCharToMultiByte(CP_UTF8,WC_ERR_INVALID_CHARS,text.data(),static_cast<int>(text.size()),nullptr,0,nullptr,nullptr);
    check(size>0,"UTF-8 conversion");
    std::string result(static_cast<size_t>(size),'\0');
    check(WideCharToMultiByte(CP_UTF8,WC_ERR_INVALID_CHARS,text.data(),static_cast<int>(text.size()),result.data(),size,nullptr,nullptr)==size,"UTF-8 conversion");
    return result;
}
#ifdef VT7PTY_H3_DIAGNOSTIC
inline std::string utf8(std::u32string_view text) {
    std::wstring wide;
    for(char32_t cp:text) {
        if(cp<=0xffff && (cp<0xd800 || cp>0xdfff)) wide+=static_cast<wchar_t>(cp);
        else if(cp>=0x10000 && cp<=0x10ffff) {
            cp-=0x10000; wide+=static_cast<wchar_t>(0xd800+(cp>>10)); wide+=static_cast<wchar_t>(0xdc00+(cp&1023));
        } else wide+=L'\ufffd';
    }
    return utf8(std::wstring_view(wide));
}
#endif
inline std::wstring imagePath(HANDLE process) {
    std::vector<wchar_t> buffer(32768); DWORD length=static_cast<DWORD>(buffer.size());
    check(QueryFullProcessImageNameW(process,0,buffer.data(),&length),"process image");
    return {buffer.data(),length};
}

struct Activity {
    Handle stop,thread;
    volatile LONG writes=0,error=0;
    static DWORD WINAPI run(void *value) {
        auto &self=*static_cast<Activity *>(value);
        while(WaitForSingleObject(self.stop.value,40)==WAIT_TIMEOUT) {
            const auto text=L"H3_ACTIVITY_"+std::to_wstring(InterlockedCompareExchange(&self.writes,0,0))+L"\r\n"; DWORD count=0;
            if(!WriteConsoleW(GetStdHandle(STD_OUTPUT_HANDLE),text.data(),static_cast<DWORD>(text.size()),&count,nullptr) || count!=text.size()) {
                InterlockedExchange(&self.error,static_cast<LONG>(GetLastError())); return 1;
            }
            InterlockedIncrement(&self.writes);
        }
        return 0;
    }
    void start() {
        check(thread.value==INVALID_HANDLE_VALUE,"one activity writer");
        stop.reset(CreateEventW(nullptr,TRUE,FALSE,nullptr)); check(stop.value!=nullptr,"activity stop");
        thread.reset(CreateThread(nullptr,0,run,this,0,nullptr)); check(thread.value!=nullptr,"owned activity writer");
    }
    void finish() {
        if(thread.value==INVALID_HANDLE_VALUE) return;
        SetEvent(stop.value); check(WaitForSingleObject(thread.value,5000)==WAIT_OBJECT_0,"activity writer stopped"); thread.reset();
    }
    ~Activity() { if(thread.value!=INVALID_HANDLE_VALUE) { SetEvent(stop.value); if(WaitForSingleObject(thread.value,5000)!=WAIT_OBJECT_0) ExitProcess(124); } }
};
#endif
struct Read {
    bool raw=false; DWORD keyCount=0; Key keys[16]={};
#ifdef VT7PTY_H3_GUARDS
    bool guardFence=false;
    uint32_t guardInputCount=0;
    GuardInputRecord guardInput[32]={};
#endif
#ifdef VT7PTY_H3_DIAGNOSTIC
    bool oneKey=false;
    uint32_t index=0,scanCodes[16]={};
    volatile LONG entered=0;
#endif
    volatile LONG returned=0;
    BOOL succeeded=FALSE;
    DWORD error=0,count=0;
    wchar_t text[128]={};
};
inline DWORD WINAPI readLine(void *data) {
    auto &r=*static_cast<Read *>(data);
#ifdef VT7PTY_H3_DIAGNOSTIC
    h3trace::fixtureTimeline.append(h3trace::ReadEntry,r.index,r.raw ? 1u : 0u);
    InterlockedExchange(&r.entered,1);
#endif
#ifdef VT7PTY_H3_GUARDS
    if(r.guardFence) {
        bool marker=false;
        while(r.guardInputCount<32) {
            INPUT_RECORD item={}; DWORD count=0;
            if(!ReadConsoleInputW(GetStdHandle(STD_INPUT_HANDLE),&item,1,&count) || count!=1) { r.error=GetLastError(); break; }
            auto &record=r.guardInput[r.guardInputCount++]; record.type=item.EventType;
            if(item.EventType!=KEY_EVENT) continue;
            const auto &key=item.Event.KeyEvent;
            record.down=key.bKeyDown ? 1u : 0u; record.character=key.uChar.UnicodeChar; record.virtualKey=key.wVirtualKeyCode;
            record.scan=key.wVirtualScanCode; record.modifiers=key.dwControlKeyState; record.repeat=key.wRepeatCount;
            if(key.bKeyDown && key.uChar.UnicodeChar==L'z' && key.wVirtualKeyCode=='Z') marker=true;
            if(marker && !key.bKeyDown && key.wVirtualKeyCode=='Z') { r.succeeded=TRUE; break; }
        }
        if(!r.succeeded && !r.error) r.error=ERROR_INSUFFICIENT_BUFFER;
        h3trace::fixtureTimeline.append(h3trace::ReadReturn,r.index,r.succeeded ? 1u : 0u,r.guardInputCount,r.error);
        InterlockedExchange(&r.returned,1); return 0;
    }
#endif
    if(r.raw) {
        unsigned int characters=0;
        while(characters<2) {
            INPUT_RECORD item={}; DWORD count=0;
            if(!ReadConsoleInputW(GetStdHandle(STD_INPUT_HANDLE),&item,1,&count) || count!=1) { r.error=GetLastError(); InterlockedExchange(&r.returned,1); return 0; }
            if(item.EventType==KEY_EVENT && item.Event.KeyEvent.bKeyDown) {
                const auto &key=item.Event.KeyEvent;
                if(r.keyCount==16) { r.error=ERROR_INSUFFICIENT_BUFFER; InterlockedExchange(&r.returned,1); return 0; }
#ifdef VT7PTY_H3_DIAGNOSTIC
                r.scanCodes[r.keyCount]=key.wVirtualScanCode;
#endif
                r.keys[r.keyCount++]={key.uChar.UnicodeChar,key.wVirtualKeyCode,key.dwControlKeyState,key.wRepeatCount};
                if(key.uChar.UnicodeChar) ++characters;
#ifdef VT7PTY_H3_DIAGNOSTIC
                if(r.oneKey) break;
#endif
            }
        }
        r.succeeded=TRUE;
#ifdef VT7PTY_H3_DIAGNOSTIC
        h3trace::fixtureTimeline.append(h3trace::ReadReturn,r.index,1,r.keyCount,r.error);
#endif
        InterlockedExchange(&r.returned,1); return 0;
    }
    SetLastError(ERROR_SUCCESS);
    r.succeeded=ReadConsoleW(GetStdHandle(STD_INPUT_HANDLE),r.text,127,&r.count,nullptr);
    r.error=GetLastError();
#ifdef VT7PTY_H3_DIAGNOSTIC
    h3trace::fixtureTimeline.append(h3trace::ReadReturn,r.index,r.succeeded ? 1u : 0u,r.count,r.error);
#endif
    InterlockedExchange(&r.returned,1); return 0;
}
inline volatile LONG controls=0;
inline volatile LONG lastControl=-1;
inline BOOL WINAPI controlHandler(DWORD type) {
    if(type!=CTRL_C_EVENT) return FALSE;
#ifdef VT7PTY_H3_DIAGNOSTIC
    const auto index=static_cast<uint32_t>(InterlockedCompareExchange(&h3trace::fixtureReadIndex,0,0));
    h3trace::fixtureTimeline.append(h3trace::HandlerEntry,index,type);
#endif
    InterlockedExchange(&lastControl,static_cast<LONG>(type));
    const auto count=InterlockedIncrement(&controls);
#ifdef VT7PTY_H3_DIAGNOSTIC
    h3trace::fixtureTimeline.append(h3trace::HandlerExit,index,type,static_cast<uint32_t>(count));
#else
    (void)count;
#endif
    return TRUE;
}
inline int fixture(const std::wstring &name,DWORD expectedServer) {
    Handle pipe(CreateFileW(name.c_str(),GENERIC_READ|GENERIC_WRITE,0,nullptr,OPEN_EXISTING,FILE_FLAG_OVERLAPPED,nullptr));
    check(pipe.value!=INVALID_HANDLE_VALUE,"fixture pipe");
    ULONG server=0; check(GetNamedPipeServerProcessId(pipe.value,&server) && server==expectedServer,"fixture server");
    check(SetConsoleMode(GetStdHandle(STD_INPUT_HANDLE),ENABLE_EXTENDED_FLAGS|ENABLE_PROCESSED_INPUT|ENABLE_LINE_INPUT|ENABLE_ECHO_INPUT),"fixture mode");
    check(FlushConsoleInputBuffer(GetStdHandle(STD_INPUT_HANDLE)),"fixture input barrier");
    #ifdef VT7PTY_H3_GUARDS
    check(SetConsoleCtrlHandler(nullptr,FALSE),"fixture explicitly opts into Ctrl+C");
#endif
    check(SetConsoleCtrlHandler(controlHandler,TRUE),"fixture control handler");
#ifdef VT7PTY_H3_DIAGNOSTIC
    bool handlerRegistered=true;
    h3trace::fixtureTimeline.append(h3trace::FixtureReady);
#endif
    Read read; Handle thread,shell,shellThread,shellJob; uint32_t readIndex=0;
#ifdef VT7PTY_H3_FREEZE
    Activity activity;
#endif
    // Preserve reader storage until completion, or exit this owned fixture.
    try {
        const ULONGLONG deadline=GetTickCount64()+
#ifdef VT7PTY_H3_FREEZE
            480000;
#else
            240000;
#endif
        while(GetTickCount64()<deadline) {
            DWORD available=0; check(PeekNamedPipe(pipe.value,nullptr,0,nullptr,&available,nullptr),"fixture command peek");
            if(!available) { Sleep(2); continue; }
            char command=0; transfer(pipe.value,&command,1,false);
            if(command=='Q') {
#ifdef VT7PTY_H3_FREEZE
                activity.finish();
#endif
                check(thread.value==INVALID_HANDLE_VALUE || WaitForSingleObject(thread.value,1000)==WAIT_OBJECT_0,"read completed before quit");
                check(shell.value==INVALID_HANDLE_VALUE || WaitForSingleObject(shell.value,0)==WAIT_OBJECT_0,"shell exited before fixture");
                return 37;
            }
            if(command=='R' || command=='U'
#ifdef VT7PTY_H3_DIAGNOSTIC
                || command=='K'
#ifdef VT7PTY_H3_GUARDS
                || command=='F'
#endif
#endif
                ) {
                if(thread.value!=INVALID_HANDLE_VALUE) {
                    check(WaitForSingleObject(thread.value,0)==WAIT_OBJECT_0,"previous reader completed");
                    thread.reset(); read=Read{};
                }
                ++readIndex;
                read.raw=command=='U';
#ifdef VT7PTY_H3_DIAGNOSTIC
                read.raw=read.raw || command=='K'; read.oneKey=command=='K'; read.index=readIndex;
#ifdef VT7PTY_H3_GUARDS
                read.guardFence=command=='F';
#endif
                InterlockedExchange(&h3trace::fixtureReadIndex,static_cast<LONG>(readIndex));
#endif
#ifdef VT7PTY_H3_GUARDS
                if(!read.guardFence) // ReadConsoleInput observes the existing queue without changing modes or flushing.
#endif
                check(SetConsoleMode(GetStdHandle(STD_INPUT_HANDLE),read.raw ? ENABLE_EXTENDED_FLAGS : ENABLE_EXTENDED_FLAGS|ENABLE_PROCESSED_INPUT|ENABLE_LINE_INPUT|ENABLE_ECHO_INPUT),"read mode");
                constexpr wchar_t marker[]=L"HOST_INPUT>"; DWORD count=0;
                check(WriteConsoleW(GetStdHandle(STD_OUTPUT_HANDLE),marker,11,&count,nullptr) && count==11,"input marker");
                thread.reset(CreateThread(nullptr,0,readLine,&read,0,nullptr)); check(thread.value!=nullptr,"reader thread");
            } else if(command=='D') {
                check(SetConsoleCtrlHandler(controlHandler,FALSE),"default handler control");
#ifdef VT7PTY_H3_DIAGNOSTIC
                handlerRegistered=false;
#endif
            } else if(command=='M' || command=='W'
#ifdef VT7PTY_H3_GUARDS
                || command=='P' || command=='L'
#endif
                ) {
                check(shell.value==INVALID_HANDLE_VALUE && thread.value==INVALID_HANDLE_VALUE,"idle shell fixture");
                wchar_t system[MAX_PATH]={}; check(GetSystemDirectoryW(system,MAX_PATH)>0,"system path");
                std::wstring exe=std::wstring(system)+(command=='M' ? L"\\cmd.exe" : L"\\WindowsPowerShell\\v1.0\\powershell.exe");
                auto line=quote(exe)+(command=='M' ? L" /D /Q /K \"prompt H3_CMD$G\"" :
                    L" -NoLogo -NoProfile -NoExit -Command \"Remove-Module PSReadLine -ErrorAction SilentlyContinue; function global:prompt { 'H3_PS>' }\"");
#ifdef VT7PTY_H3_DIAGNOSTIC
                if(command=='W') line=quote(exe)+L" -NoLogo -NoProfile -NoExit -Command \"Remove-Module PSReadLine -ErrorAction SilentlyContinue; $global:VT7PtyH3Generation=0; function global:prompt { $global:VT7PtyH3Generation++; 'H3_PS['+$global:VT7PtyH3Generation.ToString()+']>' }\"";
#endif
#ifdef VT7PTY_H3_GUARDS
                if(command=='P' || command=='L') {
                    exe=guardEnvironment(L"VT7PTY_H3_PWSH");
                    const auto literal=[](const std::wstring &s) { std::wstring q=L"'"; for(wchar_t ch:s) {q+=ch;if(ch==L'\'')q+=ch;} return q+L"'"; };
                    std::wstring boot=L"Remove-Module PSReadLine -ErrorAction SilentlyContinue; ";
                    if(command=='L') {
                        const auto module=exe.substr(0,exe.find_last_of(L'\\'))+L"\\Modules\\PSReadLine\\PSReadLine.psd1";
                        check(GetFileAttributesW(module.c_str())!=INVALID_FILE_ATTRIBUTES,"bundled PSReadLine");
                        const auto history=guardEnvironment(L"VT7PTY_H3_GUARD_OUTPUT")+L"\\psreadline-"+std::to_wstring(GetCurrentProcessId())+L".txt";
                        boot=L"Import-Module "+literal(module)+L" -ErrorAction Stop; Set-PSReadLineOption -HistorySaveStyle SaveNothing -HistorySavePath "+
                            literal(history)+L" -PredictionSource None -EditMode Windows; ";
                    }
                    boot+=L"$global:VT7PtyH3Generation=0; function global:prompt { $global:VT7PtyH3Generation++; 'H3_PS['+$global:VT7PtyH3Generation.ToString()+']>' }";
                    line=quote(exe)+L" -NoLogo -NoProfile -NoExit -Command "+quote(boot);
                }
#endif
                std::vector<wchar_t> mutableLine(line.begin(),line.end()); mutableLine.push_back(0);
                STARTUPINFOW startup={}; startup.cb=sizeof(startup); PROCESS_INFORMATION process={};
                shellJob.reset(CreateJobObjectW(nullptr,nullptr)); check(shellJob.value!=nullptr,"owned shell cleanup job");
                JOBOBJECT_EXTENDED_LIMIT_INFORMATION limits={}; limits.BasicLimitInformation.LimitFlags=JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE;
                check(SetInformationJobObject(shellJob.value,JobObjectExtendedLimitInformation,&limits,sizeof(limits)),"shell job lifetime");
                check(CreateProcessW(exe.c_str(),mutableLine.data(),nullptr,nullptr,FALSE,CREATE_SUSPENDED,nullptr,nullptr,&startup,&process),"owned shell");
                shell.reset(process.hProcess); shellThread.reset(process.hThread);
                // This job starts empty and receives only the shell we just created.
                // The existing host, agent and fixture are never assigned to it.
                check(AssignProcessToJobObject(shellJob.value,shell.value),"owned shell job assignment");
                check(ResumeThread(shellThread.value)!=static_cast<DWORD>(-1),"owned shell start");
#ifdef VT7PTY_H3_DIAGNOSTIC
                h3trace::fixtureTimeline.append(h3trace::ShellStarted,readIndex,process.dwProcessId);
#endif
#ifdef VT7PTY_H3_FREEZE
            } else if(command=='B') { activity.start();
            } else if(command=='E') { activity.finish();
#endif
            } else if(command=='T') {
                const std::wstring title=L"VT7Pty dispatch research "+std::to_wstring(GetTickCount64());
                check(SetConsoleTitleW(title.c_str()),"owned console title");
            } else check(command=='I',"identity fixture command");
            IdentityPacket p;
#ifdef VT7PTY_H3_DIAGNOSTIC
            p.trace.beginQpc=h3trace::qpc();
#endif
            p.processId=GetCurrentProcessId(); p.parentId=parentProcessId(p.processId);
            p.query=hostQuery(GetCurrentProcess());
            const HWND window=GetConsoleWindow(); p.window=reinterpret_cast<uintptr_t>(window);
            DWORD windowProcess=0; p.windowThread=GetWindowThreadProcessId(window,&windowProcess); p.windowProcess=windowProcess;
            p.consoleCount=GetConsoleProcessList(p.consoleProcesses,16);
            check(p.consoleCount>0 && p.consoleCount<=16,"console process list bound");
            if(InterlockedCompareExchange(&read.returned,0,0)) {
                p.readReturned=1; p.readSucceeded=read.succeeded ? 1u : 0u;
                p.readError=read.error; p.textCount=read.count;
                check(read.count<=127,"read text bound"); std::copy_n(read.text,read.count,p.text);
            }
            p.controlCount=static_cast<uint32_t>(InterlockedCompareExchange(&controls,0,0));
            p.controlType=static_cast<uint32_t>(InterlockedCompareExchange(&lastControl,0,0));
            p.readIndex=readIndex;
            DWORD inputMode=0; check(GetConsoleMode(GetStdHandle(STD_INPUT_HANDLE),&inputMode),"observed input mode"); p.inputMode=inputMode;
            if(p.readReturned) { p.keyCount=read.keyCount; std::copy_n(read.keys,read.keyCount,p.keys); }
#ifdef VT7PTY_H3_GUARDS
            if(p.readReturned) {p.guardInputCount=read.guardInputCount;std::copy_n(read.guardInput,read.guardInputCount,p.guardInput);}
#endif
            if(shell.value!=INVALID_HANDLE_VALUE) {
                p.shellId=GetProcessId(shell.value); p.shellAlive=WaitForSingleObject(shell.value,0)==WAIT_TIMEOUT ? 1u : 0u;
                DWORD shellExit=0; check(GetExitCodeProcess(shell.value,&shellExit),"owned shell exit"); p.shellExit=shellExit;
            }
#ifdef VT7PTY_H3_DIAGNOSTIC
            h3trace::observe(h3trace::fixtureTimeline,p.trace);
            p.trace.keyScanCount=p.keyCount; std::copy_n(read.scanCodes,p.keyCount,p.trace.keyScanCodes);
            p.trace.readEntered=static_cast<uint32_t>(InterlockedCompareExchange(&read.entered,0,0));
            p.trace.customHandlerRegistered=handlerRegistered ? 1u : 0u;
            DWORD inputCount=0; if(GetNumberOfConsoleInputEvents(GetStdHandle(STD_INPUT_HANDLE),&inputCount)) p.trace.inputRecords=inputCount;
            else p.trace.inputError=GetLastError();
            CONSOLE_SELECTION_INFO selection={};
            if(GetConsoleSelectionInfo(&selection)) p.trace.selectionFlags=selection.dwFlags; else p.trace.selectionError=GetLastError();
#endif
#ifdef VT7PTY_H3_DIAGNOSTIC
            p.trace.qpc=h3trace::qpc(); // End of the public observation interval.
#endif
#ifdef VT7PTY_H3_FREEZE
            p.activityWrites=static_cast<uint32_t>(InterlockedCompareExchange(&activity.writes,0,0));
            p.activityError=static_cast<uint32_t>(InterlockedCompareExchange(&activity.error,0,0));
            CONSOLE_SCREEN_BUFFER_INFO screen={}; check(GetConsoleScreenBufferInfo(GetStdHandle(STD_OUTPUT_HANDLE),&screen),"activity screen observation");
            p.columns=static_cast<uint32_t>(screen.srWindow.Right-screen.srWindow.Left+1); p.rows=static_cast<uint32_t>(screen.srWindow.Bottom-screen.srWindow.Top+1);
            p.trace.qpc=h3trace::qpc();
#endif
            transfer(pipe.value,&p,sizeof(p),true);
        }
        throw std::runtime_error("fixture deadline");
    } catch(...) {
        if(shell.value!=INVALID_HANDLE_VALUE && WaitForSingleObject(shell.value,0)==WAIT_TIMEOUT) {
            TerminateProcess(shell.value,124); WaitForSingleObject(shell.value,5000);
        }
        ExitProcess(124); // Closing the fixture also closes its owned-shell job.
    }
}
class Session {
    vt7pty_t *pty_=nullptr;
    Handle input_,output_,pipe_;
    bool closed_=false;
public:
    Handle agent,child,host,descendant;
#ifdef VT7PTY_H3_GUARDS
    Handle guardChild;
#endif
    VtState state{80,25};
    std::string raw;
    uint64_t lineFeeds=0;
    bool cleanupOkay=true;
#ifdef VT7PTY_H3_FREEZE
    std::unique_ptr<h3freeze::Client> gate;
#endif
#ifdef VT7PTY_H3_DIAGNOSTIC
    struct OutputFrame { uint64_t qpc,offset,lineFeeds; int x,y; uint32_t epoch; std::u32string row; };
    std::vector<OutputFrame> frames;
    bool frameOverflow=false,lastPrompt=false,haveObservation=false;
    uint32_t promptEpoch=0;
    uint64_t lastPromptLineFeeds=0;
    IdentityPacket lastObservation;
    std::string outputTraceJson() const {
        std::string result="{\"SchemaVersion\":1,\"ObserverEpochIsReadinessProof\":false,\"Overflow\":"+std::string(frameOverflow ? "true" : "false")+
            ",\"Frequency\":"+h3trace::u64(h3trace::frequency())+",\"Frames\":[";
        for(size_t i=0;i<frames.size();++i) {
            if(i) result+=','; const auto &f=frames[i];
            result+="{\"Qpc\":"+h3trace::u64(f.qpc)+",\"OutputBytes\":"+std::to_string(f.offset)+",\"LineFeeds\":"+std::to_string(f.lineFeeds)+
                ",\"X\":"+std::to_string(f.x)+",\"Y\":"+std::to_string(f.y)+",\"ObservedPromptEpoch\":"+std::to_string(f.epoch)+",\"Row\":"+json(utf8(f.row))+"}";
        }
        return result+"]}";
    }
#endif
    explicit Session(unsigned int index,const std::wstring &researchOutput=L"") {
#ifndef VT7PTY_H3_FREEZE
        (void)researchOutput;
#endif
        try {
            auto *config=vt7pty_config_new(0,nullptr); check(config!=nullptr,"config");

#ifdef VT7PTY_H3_FREEZE

#else
            pty_=vt7pty_open(config,nullptr);
#endif
            vt7pty_config_free(config); check(pty_!=nullptr,"open session");
            check(DuplicateHandle(GetCurrentProcess(),vt7pty_agent_process(pty_),GetCurrentProcess(),&agent.value,0,FALSE,DUPLICATE_SAME_ACCESS),"agent handle");
            input_.reset(CreateFileW(vt7pty_conin_name(pty_),GENERIC_WRITE,0,nullptr,OPEN_EXISTING,FILE_FLAG_OVERLAPPED,nullptr));
            output_.reset(CreateFileW(vt7pty_conout_name(pty_),GENERIC_READ,0,nullptr,OPEN_EXISTING,FILE_FLAG_OVERLAPPED,nullptr));
            check(input_.value!=INVALID_HANDLE_VALUE && output_.value!=INVALID_HANDLE_VALUE,"session pipes");
#ifdef VT7PTY_H3_FREEZE
            gate=std::make_unique<h3freeze::Client>(agent.value,nonce,researchOutput);
#endif
            const std::wstring name=L"\\\\.\\pipe\\VT7Pty-host-cancellation-"+std::to_wstring(GetCurrentProcessId())+L"-"+std::to_wstring(GetTickCount64())+L"-"+std::to_wstring(index);
            pipe_.reset(privateServer(name));
            const std::wstring exe=imagePath(GetCurrentProcess());
            const std::wstring command=quote(exe)+L" --fixture "+quote(name)+L" "+std::to_wstring(GetCurrentProcessId());
            auto *spawn=vt7pty_spawn_config_new(VT7PTY_SPAWN_FLAG_AUTO_SHUTDOWN,exe.c_str(),command.c_str(),nullptr,nullptr,nullptr);
            check(spawn!=nullptr,"spawn config");
            const BOOL okay=vt7pty_spawn(pty_,spawn,&child.value,nullptr,nullptr,nullptr);
            vt7pty_spawn_config_free(spawn); check(okay!=FALSE,"spawn fixture");
            connect(pipe_.value);
            ULONG peer=0; check(GetNamedPipeClientProcessId(pipe_.value,&peer) && peer==GetProcessId(child.value),"owned pipe client");
            check(_wcsicmp(imagePath(child.value).c_str(),exe.c_str())==0,"fixture image");
        } catch(...) { cleanup(); throw; }
    }
    ~Session() { cleanup(); }
    IdentityPacket observe(char command='I') {
        transfer(pipe_.value,&command,1,true);
        IdentityPacket p; transfer(pipe_.value,&p,sizeof(p),false);
        check(p.magic==0x33494856 && p.schema==
#if defined(VT7PTY_H3_GUARDS)
            5
#elif defined(VT7PTY_H3_COORDINATED)
            4
#elif defined(VT7PTY_H3_FREEZE)
            3
#else
            1
#endif
            #ifdef VT7PTY_H3_GUARDS
            && p.guardInputCount<=32
#endif
            && p.processId==GetProcessId(child.value) &&
            p.parentId==GetProcessId(agent.value) && p.window!=0 && p.windowThread!=0 &&
            p.consoleCount>0 && p.consoleCount<=16 && p.readReturned<=1 && p.readSucceeded<=1 && p.textCount<=127 && p.keyCount<=16 && p.controlCount<=128,"identity packet");
        const auto contains=[&](DWORD id) { return std::find(p.consoleProcesses,p.consoleProcesses+p.consoleCount,id)!=p.consoleProcesses+p.consoleCount; };
        check(contains(p.processId) && contains(p.parentId) && (p.consoleCount==2 || (p.consoleCount==3 && p.shellId && contains(p.shellId))
#ifdef VT7PTY_H3_GUARDS
            || (p.consoleCount==4 && p.shellId && contains(p.shellId) && guardChild.value && guardChild.value!=INVALID_HANDLE_VALUE && contains(GetProcessId(guardChild.value)))
#endif
            ),"owned console members");
#ifdef VT7PTY_H3_DIAGNOSTIC
        lastObservation=p; haveObservation=true;
#endif
        return p;
    }
    void input(std::string_view text) { transfer(input_.value,const_cast<char *>(text.data()),static_cast<DWORD>(text.size()),true); }
    void pump() {
        for(unsigned int n=0;n<64;++n) {
            DWORD available=0;
            if(!PeekNamedPipe(output_.value,nullptr,0,nullptr,&available,nullptr)) {
                check(GetLastError()==ERROR_BROKEN_PIPE || GetLastError()==ERROR_PIPE_NOT_CONNECTED,"output peek"); return;
            }
            if(!available) return;
            char buffer[4096]; DWORD size=std::min<DWORD>(available,sizeof(buffer));
            transfer(output_.value,buffer,size,false);
            check(raw.size()+size<=1024*1024,"output size"); raw.append(buffer,size); lineFeeds+=std::count(buffer,buffer+size,'\n'); state.feed({buffer,size});
#ifdef VT7PTY_H3_DIAGNOSTIC
            auto row=state.row(state.y()); while(!row.empty() && row.back()==U' ') row.pop_back();
            const bool prompt=row==U"H3_CMD>" || row==U"H3_PS>" ||
                (row.starts_with(U"H3_PS[") && row.ends_with(U"]>")) || (row.starts_with(U"H3_CMD_") && row.ends_with(U">"));
            if(prompt && (!lastPrompt || lineFeeds!=lastPromptLineFeeds)) ++promptEpoch;
            if(prompt) lastPromptLineFeeds=lineFeeds; lastPrompt=prompt;
            if(frames.size()<512) frames.push_back({h3trace::qpc(),raw.size(),lineFeeds,state.x(),state.y(),promptEpoch,std::move(row)});
            else frameOverflow=true;
#endif
            for(const auto &reply:state.takeReplies()) input(reply);
        }
    }
    bool until(const std::function<bool()> &condition,DWORD timeout=5000) {
        const ULONGLONG end=GetTickCount64()+timeout;
        do { pump(); if(condition()) return true; Sleep(2); } while(GetTickCount64()<end);
        return false;
    }
    void quit() {
#ifdef VT7PTY_H3_DIAGNOSTIC
        observe(); // Final cumulative timeline before the fixture exits normally.
#endif
#ifdef VT7PTY_H3_FREEZE
        if(gate) { check(gate->finish(),"agent trace final collection"); gate.reset(); }
#endif
        char command='Q'; transfer(pipe_.value,&command,1,true);
        check(until([&] { return WaitForSingleObject(child.value,0)==WAIT_OBJECT_0; }),"fixture quit");
        DWORD code=0; check(GetExitCodeProcess(child.value,&code) && code==37,"fixture exit status");
        check(cleanup(),"session cleanup");
    }
    bool cleanup() noexcept {
        if(closed_) return cleanupOkay;
        closed_=true;
#ifdef VT7PTY_H3_FREEZE
        if(gate) { cleanupOkay=gate->finish() && cleanupOkay; gate.reset(); }
#endif
        pipe_.reset(); input_.reset();
        if(descendant.value && descendant.value!=INVALID_HANDLE_VALUE && WaitForSingleObject(descendant.value,1000)!=WAIT_OBJECT_0) {
            cleanupOkay=false; TerminateProcess(descendant.value,125); WaitForSingleObject(descendant.value,5000);
        }
        if(pty_) { vt7pty_free(pty_); pty_=nullptr; }
        for(HANDLE process:{child.value,agent.value}) {
            if(process==nullptr || process==INVALID_HANDLE_VALUE) continue;
            if(WaitForSingleObject(process,5000)!=WAIT_OBJECT_0) {
                cleanupOkay=false; TerminateProcess(process,125); WaitForSingleObject(process,5000);
            }
        }
        #ifdef VT7PTY_H3_GUARDS
        if(guardChild.value && guardChild.value!=INVALID_HANDLE_VALUE && WaitForSingleObject(guardChild.value,5000)!=WAIT_OBJECT_0) {
            cleanupOkay=false; TerminateProcess(guardChild.value,125); WaitForSingleObject(guardChild.value,5000);
        }
#endif
        output_.reset(); return cleanupOkay;
    }
};
struct TokenIdentity {
    std::vector<unsigned char> user;
    TOKEN_STATISTICS statistics={};
    DWORD integrity=0,session=0;
    explicit TokenIdentity(HANDLE process) {
        Handle token; check(OpenProcessToken(process,TOKEN_QUERY,&token.value),"process token");
        DWORD size=0; GetTokenInformation(token.value,TokenUser,nullptr,0,&size);
        check(size>0 && size<65536,"token size"); user.resize(size);
        check(GetTokenInformation(token.value,TokenUser,user.data(),size,&size),"token user");
        check(GetTokenInformation(token.value,TokenStatistics,&statistics,sizeof(statistics),&size),"token statistics");
        check(GetTokenInformation(token.value,TokenSessionId,&session,sizeof(session),&size),"token session");
        size=0; GetTokenInformation(token.value,TokenIntegrityLevel,nullptr,0,&size);
        check(size>0 && size<65536,"integrity size"); std::vector<unsigned char> storage(size);
        check(GetTokenInformation(token.value,TokenIntegrityLevel,storage.data(),size,&size),"integrity level");
        auto *sid=reinterpret_cast<TOKEN_MANDATORY_LABEL *>(storage.data())->Label.Sid;
        check(IsValidSid(sid) && *GetSidSubAuthorityCount(sid)>0,"integrity SID");
        integrity=*GetSidSubAuthority(sid,*GetSidSubAuthorityCount(sid)-1);
    }
    bool same(const TokenIdentity &other) const {
        return EqualSid(reinterpret_cast<const TOKEN_USER *>(user.data())->User.Sid,
            reinterpret_cast<const TOKEN_USER *>(other.user.data())->User.Sid) &&
            statistics.AuthenticationId.LowPart==other.statistics.AuthenticationId.LowPart &&
            statistics.AuthenticationId.HighPart==other.statistics.AuthenticationId.HighPart &&
            integrity==other.integrity && session==other.session;
    }
};
inline uint64_t creation(HANDLE process) {
    FILETIME c={},e={},k={},u={}; check(GetProcessTimes(process,&c,&e,&k,&u),"process creation time");
    return (static_cast<uint64_t>(c.dwHighDateTime)<<32)|c.dwLowDateTime;
}
inline std::string guiThreads(DWORD pid) {
    Handle snapshot(CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD,0)); check(snapshot.value!=INVALID_HANDLE_VALUE,"thread snapshot");
    THREADENTRY32 entry={}; entry.dwSize=sizeof(entry); check(Thread32First(snapshot.value,&entry),"first thread");
    std::string result="["; unsigned int count=0;
    do {
        if(entry.th32OwnerProcessID!=pid) continue;
        check(++count<=128,"host thread bound");
        if(count>1) result+=',';
        Handle thread(OpenThread(THREAD_QUERY_LIMITED_INFORMATION,FALSE,entry.th32ThreadID));
        const DWORD openError=thread.value ? 0 : GetLastError();
        const bool belongs=thread.value && GetProcessIdOfThread(thread.value)==pid;
        GUITHREADINFO gui={}; gui.cbSize=sizeof(gui); SetLastError(ERROR_SUCCESS);
        const BOOL okay=belongs ? GetGUIThreadInfo(entry.th32ThreadID,&gui) : FALSE;
        const DWORD error=belongs ? (okay ? 0 : GetLastError()) : openError;
        result+="{\"ThreadId\":"+std::to_string(entry.th32ThreadID)+",\"OwnedThreadVerified\":"+(belongs ? "true" : "false")+
            ",\"GuiInfoSucceeded\":"+(okay ? "true" : "false")+",\"Error\":"+std::to_string(error)+"}";
    } while(Thread32Next(snapshot.value,&entry));
    check(GetLastError()==ERROR_NO_MORE_FILES,"thread enumeration");
    return result+"]";
}

inline bool observeIdentity(Session &s,const std::wstring &output,const wchar_t *label) {
    const IdentityPacket packet=s.observe();
    const Query child=hostQuery(s.child.value),agent=hostQuery(s.agent.value);
    std::string record="{\"Fixture\":"+packetJson(packet)+",\"ChildQuery\":"+queryJson(child)+",\"AgentQuery\":"+queryJson(agent);
    bool eligible=false; std::string error;
    try {
        check(WaitForSingleObject(s.child.value,0)==WAIT_TIMEOUT && WaitForSingleObject(s.agent.value,0)==WAIT_TIMEOUT,"live owned clients");
        check(agrees(packet.query,child) && agrees(child,agent),"consistent host identity");
        // libuv v1.44.2 tty.c also removes the low two flag bits before using the PID.
        const DWORD pid=static_cast<DWORD>(child.raw&~ULONG_PTR{3});
        check(pid!=GetCurrentProcessId() && pid!=packet.processId && pid!=packet.parentId,"distinct host process");
        if(s.host.value==INVALID_HANDLE_VALUE) s.host.reset(OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION|SYNCHRONIZE,FALSE,pid));
        if(!s.host.value) throw std::runtime_error("host query access denied or unavailable: "+std::to_string(GetLastError()));
        check(GetProcessId(s.host.value)==pid,"retained host query handle");
        check(WaitForSingleObject(s.host.value,0)==WAIT_TIMEOUT,"live host");
        const std::wstring path=imagePath(s.host.value);
        wchar_t system[MAX_PATH]={}; const UINT length=GetSystemDirectoryW(system,MAX_PATH);
        check(length>0 && length<MAX_PATH,"system path");
        check(_wcsicmp(path.c_str(),(std::wstring(system)+L"\\conhost.exe").c_str())==0,"System32 host image");
        const TokenIdentity ours(GetCurrentProcess()),host(s.host.value),client(s.child.value),owner(s.agent.value);
        check(ours.same(host) && ours.same(client) && ours.same(owner),"same user logon integrity and session");
        BOOL wow=TRUE; check(IsWow64Process(s.host.value,&wow) && !wow,"native x64 host");
        check(creation(s.host.value)<=creation(s.child.value),"host predates fixture");
        record+=",\"HostProcessId\":"+std::to_string(pid)+",\"HostImage\":"+json(utf8(path))+
            ",\"HostCreationTime\":"+json(std::to_string(creation(s.host.value)))+
            ",\"SessionId\":"+std::to_string(host.session)+",\"IntegrityRid\":"+std::to_string(host.integrity)+
            ",\"SameUserLogonIntegritySession\":true,\"NativeX64\":true,\"Threads\":"+guiThreads(pid);
        check(agrees(hostQuery(s.child.value),child) && agrees(hostQuery(s.agent.value),agent),"stable host association");
        eligible=true;
    } catch(const std::exception &e) { error=e.what(); }
    record+=",\"IdentityEligible\":"+std::string(eligible ? "true" : "false")+
        ",\"DispatchThreadVerified\":false,\"Error\":"+json(error)+"}";
    writeFile(output+L"\\"+label+L".json",record);
    return eligible;
}
inline void oracle() {
    const Query good{0,sizeof(ULONG_PTR),400};
    for(ULONG_PTR flags=0;flags<4;++flags) {
        const Query q{0,sizeof(ULONG_PTR),400|flags};
        check(valid(q) && agrees(q,q) && (q.raw&~ULONG_PTR{3})==400,"oracle flagged PID");
    }
    std::array<Query,5> bad={Query{static_cast<LONG>(0xc0000003u),sizeof(ULONG_PTR),400},
        Query{0,4,400},Query{0,sizeof(ULONG_PTR),0},Query{0,sizeof(ULONG_PTR),0x100000000ull},Query{0,sizeof(ULONG_PTR),3}};
    for(const auto &q:bad) check(!valid(q) && !agrees(good,q),"oracle invalid query");
    check(!agrees(good,Query{0,sizeof(ULONG_PTR),404}),"oracle different host");
    puts("Host identity oracle: 10 checks passed");
}
inline void startRead(Session &s,const std::string &text) {
    check(!s.observe('R').readReturned,"read starts pending"); s.input(text);
    std::u32string expected=U"HOST_INPUT>"; for(unsigned char c:text) expected+=c;
    check(s.until([&] { return s.state.contains(expected); }),"pending input echo");
    check(!s.observe().readReturned,"echoed read still pending");
}
inline void finishRead(Session &s,const std::wstring &text,const std::wstring &output,const wchar_t *label) {
    s.input("\r"); IdentityPacket p;
    check(s.until([&] { p=s.observe(); return p.readReturned!=0; }),"ordinary read completion");
    writeFile(output+L"\\"+label+L".json",packetJson(p));
    check(p.readSucceeded && p.readError==0 && std::wstring(p.text,p.textCount)==text+L"\r\n","isolated read text");
}
