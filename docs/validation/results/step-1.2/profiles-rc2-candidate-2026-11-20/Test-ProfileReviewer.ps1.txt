[CmdletBinding()]
param([Parameter(Mandatory=$true)][string]$NativeDirectory,
    [Parameter(Mandatory=$true)][string]$ManualDirectory,
    [Parameter(Mandatory=$true)][string]$OutputDirectory)
$ErrorActionPreference='Stop'
Set-StrictMode -Version 2.0
$OutputDirectory=[IO.Path]::GetFullPath($OutputDirectory)
$artifactRoot=[IO.Path]::GetFullPath((Join-Path $PSScriptRoot '..\..\artifacts')).TrimEnd('\')
if(-not $OutputDirectory.StartsWith($artifactRoot+'\',[StringComparison]::OrdinalIgnoreCase) -or
    (Test-Path -LiteralPath $OutputDirectory)){throw 'Use a fresh control directory under repository artifacts.'}
[void][IO.Directory]::CreateDirectory($OutputDirectory)
$fixture=Join-Path $OutputDirectory 'fixture';$manual=Join-Path $OutputDirectory 'manual'
Copy-Item -LiteralPath $NativeDirectory -Destination $fixture -Recurse
Copy-Item -LiteralPath $ManualDirectory -Destination $manual -Recurse
$reviewer=Join-Path $PSScriptRoot 'Test-ProfileEvidence.ps1'
$baseline=& $reviewer -Directory $fixture
$smoke=& $reviewer -Directory $manual -AllowSynthetic
if(-not $smoke.SyntheticSmoke -or $smoke.OrdinaryProfilesQualified){throw 'Control input must be a nonqualifying manual smoke.'}
$controls=@(
    @{Name='wrong-runtime';Root=$fixture;File='probe.json';Expected='native profile identity';Edit={param($v)$v.RuntimeSourceCommit=('0'*40)}},
    @{Name='false-ordinary-fixture';Root=$fixture;File='probe.json';Expected='explicit fixture scope';Edit={param($v)$v.OrdinaryUserProfiles='Pass'}},
    @{Name='false-step-acceptance';Root=$fixture;File='probe.json';Expected='explicit fixture scope';Edit={param($v)$v.Step12Accepted=$true}},
    @{Name='missing-case';Root=$fixture;File='cases.json';Expected='case inventory';Edit={param($v)$v.Cases=@($v.Cases | Select-Object -Skip 1)}},
    @{Name='changed-launch';Root=$fixture;File='wps-clean-native\shell-launch.json';Expected='controlled launch';Edit={param($v)$v.NoProfile=$false}},
    @{Name='wrong-shell-parent';Root=$fixture;File='wps-clean-native\shell-launch.json';Expected='retained session';Edit={param($v)$v.ShellParentId++}},
    @{Name='changed-profile';Root=$fixture;File='wps-clean-native\snapshot-1.json';Expected='profile files preserved';Edit={param($v)$v.Profiles[0].Present=-not $v.Profiles[0].Present;$v.Profiles[0].Sha256=('a'*64)}},
    @{Name='changed-policy';Root=$fixture;File='wps-clean-native\snapshot-1.json';Expected='execution policy preserved';Edit={param($v)$v.Policy.Process='Bypass'}},
    @{Name='wrong-precedence';Root=$fixture;File='wps-layered-native\snapshot-0.json';Expected='session identity';Edit={param($v)$v.FixtureWinner='AllUsersAllHosts'}},
    @{Name='startup-errors';Root=$fixture;File='wps-clean-native\snapshot-0.json';Expected='workflow errors';Edit={param($v)$v.StartupErrorCount=1}},
    @{Name='wrong-backend-text';Root=$fixture;File='wps-clean-native\Editing.json';Expected='command proof Editing';Edit={param($v)$v.Matched=$false}},
    @{Name='history-not-recalled';Root=$fixture;File='wps-clean-native\History.json';Expected='invocation count';Edit={param($v)$v.CheckInvocation=1}},
    @{Name='wrong-child-event';Root=$fixture;File='wps-clean-native\child-result.json';Expected='native child identity/event';Edit={param($v)$v.CtrlC=2}},
    @{Name='wrong-resize';Root=$fixture;File='wps-clean-native\snapshot-1.json';Expected='resize readback';Edit={param($v)$v.WindowWidth=99}},
    @{Name='host-not-cleaned';Root=$fixture;File='wps-clean-native\lifetime.json';Expected='retained session';Edit={param($v)$v.HostExited=$false}},
    @{Name='manual-false-profiles';Root=$manual;File='probe.json';Expected='never synthetic acceptance';Edit={param($v)$v.OrdinaryProfilesLoaded=$true}},
    @{Name='manual-private-content';Root=$manual;File='probe.json';Expected='completion/privacy';Edit={param($v)$v.ProfileContentCollected=$true}},
    @{Name='manual-no-resize';Root=$manual;File='physical-checks.json';Expected='checkpoint and resize';Edit={param($v)$v.Resizes=@()}},
    @{Name='manual-wrong-input-source';Root=$manual;File='physical-checks.json';Expected='checkpoint source';Edit={param($v)$v.Stages[1].Source='Unknown'}},
    @{Name='manual-no-edit-key';Root=$manual;File='physical-checks.json';Expected='physical editing/history';Edit={param($v)$v.KeyCountsByStage[1].'37'=0}},
    @{Name='manual-changed-editor';Root=$manual;File='session\snapshot-1.json';Expected='editor options preserved';Edit={param($v)$v.Options.EditMode='Unknown'}}
)
if($baseline.PublicationChecks -eq 3){$controls+=@(
    @{Name='replacement-while-held-failed';Root=$fixture;File='publication\result.json';Expected='held-reader publication';Edit={param($v)$v.SharedReaderReplacement=$false}},
    @{Name='stale-published-proof';Root=$fixture;File='publication\proof.json';Expected='replaced publication proofs';Edit={param($v)$v.Invocation=1}}
)}
$results=@()
foreach($control in $controls){
    $path=Join-Path $control.Root $control.File;$original=[IO.File]::ReadAllBytes($path)
    try{
        $value=[Text.Encoding]::UTF8.GetString($original) | ConvertFrom-Json
        & $control.Edit $value
        [IO.File]::WriteAllText($path,($value | ConvertTo-Json -Depth 30),[Text.UTF8Encoding]::new($false))
        $reason=''
        try{$null=& $reviewer -Directory $control.Root -AllowSynthetic}catch{$reason=$_.Exception.Message}
        if(-not $reason.Contains($control.Expected)){throw ('Missed '+$control.Name+': '+$reason)}
        $results+=@(@{Case=$control.Name;Rejected=$true;Reason=$reason})
    }finally{[IO.File]::WriteAllBytes($path,$original)}
}
$reason='';try{$null=& $reviewer -Directory $manual}catch{$reason=$_.Exception.Message}
if(-not $reason.Contains('never synthetic acceptance')){throw 'Synthetic smoke was accepted as physical input.'}
$results+=@(@{Case='synthetic-is-not-physical';Rejected=$true;Reason=$reason})
foreach($control in @(@{Name='canceled-line-executed';Root=$fixture;Path='wps-clean-native\Cancelled.json';Expected='canceled profile line'},
    @{Name='ordinary-output-collected';Root=$manual;Path='session\stream.bin';Expected='no ordinary output'})){
    $path=Join-Path $control.Root $control.Path;[IO.File]::WriteAllText($path,'{}')
    try{
        $reason='';try{$null=& $reviewer -Directory $control.Root -AllowSynthetic}catch{$reason=$_.Exception.Message}
        if(-not $reason.Contains($control.Expected)){throw ('Missed '+$control.Name+': '+$reason)}
        $results+=@(@{Case=$control.Name;Rejected=$true;Reason=$reason})
    }finally{[IO.File]::Delete($path)}
}
$path=Join-Path $fixture 'runtime-trace.jsonl';$original=[IO.File]::ReadAllBytes($path)
try{
    $text=[Text.Encoding]::UTF8.GetString($original).Replace('0ca6e3fa78481cd4df35820b9fe3988f46e0bff4',('0'*40))
    [IO.File]::WriteAllText($path,$text,[Text.UTF8Encoding]::new($false))
    $reason='';try{$null=& $reviewer -Directory $fixture}catch{$reason=$_.Exception.Message}
    if(-not $reason.Contains('trace runtime/harness/API/protocol')){throw 'Wrong runtime trace was accepted.'}
    $results+=@(@{Case='wrong-trace-runtime';Rejected=$true;Reason=$reason})
}finally{[IO.File]::WriteAllBytes($path,$original)}
$again=& $reviewer -Directory $fixture;$againSmoke=& $reviewer -Directory $manual -AllowSynthetic
if($again.Cases.Count -ne $baseline.Cases.Count -or $againSmoke.OrdinaryProfilesQualified){throw 'Control restoration changed the evidence.'}
$result=[ordered]@{Kind='ProfileReviewerControls';Status='Pass';Rejected=$results.Count;FixtureCases=$baseline.Cases.Count;ManualStages=$smoke.Stages;Controls=$results;Step12Accepted=$false}
[IO.File]::WriteAllText((Join-Path $OutputDirectory 'review.json'),($result | ConvertTo-Json -Depth 12),[Text.UTF8Encoding]::new($false))
$result
