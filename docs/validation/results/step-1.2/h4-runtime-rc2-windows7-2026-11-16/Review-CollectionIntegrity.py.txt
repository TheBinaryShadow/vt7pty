from pathlib import Path
import collections, datetime, hashlib, json, sys

root=Path(sys.argv[1])
package=Path(sys.argv[2])
shared=Path(sys.argv[3])
audit=root/"audit"
def read(path):
    return json.loads(path.read_bytes())
def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def require(condition, message):
    if not condition:
        raise RuntimeError(message)
def inventory(base, name="evidence-files.json", complete=True):
    record=read(base/name)
    seen=set()
    for item in record["Files"]:
        path=(base/item["Path"]).resolve()
        relative=path.relative_to(base.resolve()).as_posix()
        require(relative not in seen, "Duplicate inventory: "+str(path))
        seen.add(relative)
        require(path.stat().st_size==item["Bytes"] and digest(path)==item["Sha256"], "Inventory mismatch: "+str(path))
    if complete:
        actual={p.relative_to(base).as_posix() for p in base.rglob("*") if p.is_file()}
        require(actual==seen|{name}, "Incomplete inventory: "+str(base))
    return len(seen)

manifest=read(package/"manifest.json")
manifest_hash=digest(package/"manifest.json")
require(manifest_hash=="e49e33da459d6c6dc97ec118fe733b503ebdbdd4fd1360c74f41b0ef767a2bd4", "Wrong rc2 manifest")
package_count=inventory(package,"manifest.json",False)
require(package_count==213,"Wrong package file inventory")
definitions=[
    ("NESSY","NonESU","nonesu-h4-20261115-234626-c252b6cb"),
    ("TURTLE","ESU","esu-h4-20261115-234706-ce186675"),
    ("LEOPARD","Legacy","legacy-h4-20261115-234613-4dac46d7")
]
integrity=[]
metrics=[]
for machine,tier,run_id in definitions:
    run=root/run_id
    source=shared/run_id
    report=read(run/"report.json")
    require(report["ComputerName"]==machine and report["Tier"]==tier and report["RunId"]==run_id, "Wrong machine/tier/run")
    require(report["ManifestSha256"]==manifest_hash and report["Runtime"]==manifest["Runtime"] and report["Reference"]==manifest["Reference"], "Candidate identity mismatch")
    require(report["OS"]["Version"]=="6.1.7601" and report["OS"]["ServicePackMajorVersion"]==1,"Windows 7 SP1 required")
    require(not report["PreflightErrors"] and not report["H4Accepted"] and not report["MilestoneAccepted"],"Unexpected report state")
    require(len(report["Trials"])==9 and [t["Order"] for t in report["Trials"]]==list(range(1,10)),"Wrong trial order")
    top_hashes=inventory(run)
    trials=[]
    trial_hashes=0
    for trial in report["Trials"]:
        order=trial["Order"]
        role=["Control","Reference","Runtime"][(order-1)%3]
        require(trial["Role"]==role and trial["Group"]==(order-1)//3+1 and trial["CollectionStatus"]=="Completed" and not trial["TimedOut"] and not trial["Error"],"Incomplete/interleaved trial")
        trial_root=run/trial["Directory"]
        trial_hashes+=inventory(trial_root)
        native=trial_root/"native"
        probe=read(native/"probe.json")
        require(probe["Status"]==trial["NativeOutcome"]==trial["Outcome"],"Producer status changed")
        require((trial["ExitCode"]==0)==(probe["Status"]=="Pass"),"Exit status mismatch")
        case_file=native/("cases.json" if role=="Runtime" else "suite/cases.json")
        cases=read(case_file)["Cases"]
        expected=12 if role=="Control" else (26 if tier=="Legacy" else 34) if role=="Reference" else (33 if tier=="Legacy" else 41)
        passed=sum(c["Passed"] for c in cases)
        require(len(cases)==expected and ((passed==expected)==(probe["Status"]=="Pass")),"Native aggregate mismatch")
        if role=="Runtime":
            require(probe["Passed"]==passed and probe["Total"]==expected and probe["SourceCommit"]==manifest["Runtime"]["SourceCommit"],"Runtime aggregate/identity mismatch")
        failures=[c for c in cases if not c["Passed"]]
        trials.append({
            "Order":order,"Role":role,"NativeOutcome":probe["Status"],"Passed":passed,"Total":expected,
            "FailedCases":failures,
            "OwnedCleanupFalse":sum(not c["OwnedProcessesCleaned"] for c in cases),
            "HostExitFalse":sum(not c["HostExited"] for c in cases),
            "ErrorGroups":[{"Error":error,"Count":count} for error,count in collections.Counter(c["Error"] for c in failures).items()]
        })
    regression_files=list((run/"regression").glob("*.json"))
    require(len(regression_files)==1,"One regression report required")
    regression=read(regression_files[0])
    require(regression["Status"]=="Pass" and regression["DeclaredTier"]==tier and len(regression["Tests"])==30,"Regression aggregate mismatch")
    require(regression["Package"]["SourceCommit"]==manifest["Runtime"]["SourceCommit"] and regression["Package"]["ManifestSha256"]==manifest["Regression"]["ManifestSha256"],"Regression package mismatch")
    require(len({t["Name"] for t in regression["Tests"]})==30,"Duplicate regression case")
    negative_names={
        "Negative control: NEGATIVE_OUTPUT","Negative control: NEGATIVE_STATUS",
        "Negative control: NEGATIVE_ORDER","Negative control: NEGATIVE_TRUNCATION",
        "Negative control: NEGATIVE_LEAK","Negative control: timeout"
    }
    require({t["Name"] for t in regression["Tests"] if t["Name"].startswith("Negative control: ")}==negative_names,"Negative-control inventory changed")
    for test in regression["Tests"]:
        require(test["Status"]=="Pass" and not test["Failure"],"Regression oracle failed")
        if test["Name"]=="Negative control: timeout":
            require(test["TimedOut"] and test["ExitCode"] is None,"Timeout negative control not detected")
        elif test["Name"] in negative_names:
            require(test["ExitCode"]==1 and not test["TimedOut"],"Negative oracle was not rejected")
        else:
            require(test["ExitCode"]==0 and not test["TimedOut"],"Unexpected regression exit/timeout")
    require(not regression["PreflightFailures"] and not regression["CrashArtifacts"],"Regression preflight/crash evidence")
    soak=[t for t in regression["Tests"] if t["Name"]=="Session contract: SOAK 120 minutes"]
    require(len(soak)==1 and soak[0]["Arguments"]==["SOAK","120"] and soak[0]["DurationMilliseconds"]>=7200000,"Full soak not established")
    require("Session contract SOAK passed." in soak[0]["StandardOutput"],"Soak output missing")
    require(report["ManualKeyboardQualification"]=="Pending" and report["OptInSustainedWorkloadQualification"]=="Pending","Qualification boundary changed")
    if tier=="Legacy":
        require(report["PowerShellVersion"]=="2.0" and report["CLRVersion"]=="2.0.50727.8806" and not report["PowerShell7"]["Applicable"],"Legacy prerequisite profile changed")
        require(not any(u["Id"].upper()=="KB3191566" for u in report["Updates"]),"Legacy KB3191566 present")
    else:
        ps7=report["PowerShell7"]
        require(ps7["Applicable"] and ps7["EngineVersion"]=="7.2.24" and ps7["ExecutableSha256"]=="689a44bf484cef719bf5022396c2bd82ee81860029eb1152f06c4771d9b032b6" and ps7["ModuleManifestSha256"]=="4d66ce291d8fbfe71e8f565c51cb28545ad3344071107f18979b29632ee97596","Pinned PS7 profile changed")
    local_files={p.relative_to(run).as_posix():p.stat().st_size for p in run.rglob("*") if p.is_file()}
    source_files={p.relative_to(source).as_posix():p.stat().st_size for p in source.rglob("*") if p.is_file()}
    require(local_files==source_files,"Shared/snapshot names or lengths differ")
    metadata=[run/"evidence-files.json"]+[run/t["Directory"]/"evidence-files.json" for t in report["Trials"]]
    for file in metadata:
        require(digest(file)==digest(source/file.relative_to(run)),"Shared inventory differs")
    integrity.append({
        "Machine":machine,"RunId":run_id,"Files":len(local_files),"Bytes":sum(local_files.values()),
        "TopInventoryHashesVerified":top_hashes,"TrialInventoryHashesVerified":trial_hashes,
        "SharedNamesAndLengthsMatch":True,"SharedInventoryHashesVerified":len(metadata)
    })
    totals={role:{"Passed":sum(t["Passed"] for t in trials if t["Role"]==role),"Total":sum(t["Total"] for t in trials if t["Role"]==role)} for role in ["Control","Reference","Runtime"]}
    metrics.append({
        "Machine":machine,"Tier":tier,"RunId":run_id,"PowerShellVersion":report["PowerShellVersion"],"CLRVersion":report["CLRVersion"],
        "Totals":totals,"Trials":trials,"RegressionPassed":30,
        "SoakDurationMilliseconds":soak[0]["DurationMilliseconds"],"SoakOutput":soak[0]["StandardOutput"],
        "ManualKeyboardQualification":"Pending","OptInSustainedWorkloadQualification":"Pending","H4Accepted":False
    })
    print(json.dumps({"Machine":machine,"Totals":totals,"Files":len(local_files),"SoakMilliseconds":soak[0]["DurationMilliseconds"]}),flush=True)
integrity_record={"Kind":"H4Rc2ReturnedCollectionIntegrity","RecordedUtc":datetime.datetime.now(datetime.timezone.utc).isoformat(),"PackageManifestSha256":manifest_hash,"PackageFilesVerified":package_count,"Collections":integrity}
(audit/"collection-integrity.json").write_text(json.dumps(integrity_record,indent=2),encoding="utf-8")
(audit/"metrics.json").write_text(json.dumps({"Kind":"H4Rc2Windows7Metrics","Collections":metrics,"H4Accepted":False},indent=2),encoding="utf-8")
print(json.dumps({"ReturnedFiles":sum(c["Files"] for c in integrity),"ReturnedBytes":sum(c["Bytes"] for c in integrity),"AllDeclaredHashesVerified":True}),flush=True)
