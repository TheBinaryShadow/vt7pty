function Copy-ExactFile([string]$Source,[string]$Relative) {
    $sourcePath=[IO.Path]::GetFullPath($Source)
    $destination=[IO.Path]::GetFullPath((Join-Path $target $Relative))
    if(-not $destination.StartsWith($target+'\',[StringComparison]::OrdinalIgnoreCase)) {throw 'Evidence path leaves its directory.'}
    [IO.Directory]::CreateDirectory((Split-Path -Parent $destination)) | Out-Null
    $hash=(Get-FileHash -LiteralPath $sourcePath -Algorithm SHA256).Hash.ToLowerInvariant()
    if(Test-Path -LiteralPath $destination) {
        if((Get-FileHash -LiteralPath $destination -Algorithm SHA256).Hash.ToLowerInvariant() -ne $hash) {throw ('Refusing to replace different evidence: '+$Relative)}
    } else {[IO.File]::Copy($sourcePath,$destination)}
    if((Get-Item -LiteralPath $destination).Length -ne (Get-Item -LiteralPath $sourcePath).Length -or
        (Get-FileHash -LiteralPath $destination -Algorithm SHA256).Hash.ToLowerInvariant() -ne $hash) {throw ('Evidence copy mismatch: '+$Relative)}
    $records.Add([ordered]@{Path=$Relative.Replace('\','/');Bytes=(Get-Item -LiteralPath $destination).Length;Sha256=$hash})
}
function Copy-ExactTree([string]$Source,[string]$Relative,[bool]$SkipBinaries=$false) {
    $sourcePath=[IO.Path]::GetFullPath($Source).TrimEnd('\')
    foreach($file in @(Get-ChildItem -LiteralPath $sourcePath -File -Recurse)) {
        if($SkipBinaries -and $file.Extension -in @('.exe','.dll','.pdb','.lib','.exp','.obj')) {continue}
        Copy-ExactFile $file.FullName (Join-Path $Relative $file.FullName.Substring($sourcePath.Length+1))
    }
}
